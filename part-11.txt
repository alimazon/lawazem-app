  y1: string;
  x2: string;
  y2: string;
  major: boolean;
}

const TICKS: readonly Tick[] = Array.from({ length: TICK_COUNT }, (_, i) => {
  const a = (i / TICK_COUNT) * 2 * Math.PI;
  const major = i % 5 === 0;
  const r1 = major ? 129 : 135;
  const r2 = 143;
  return {
    x1: (C + r1 * Math.cos(a)).toFixed(2),
    y1: (C + r1 * Math.sin(a)).toFixed(2),
    x2: (C + r2 * Math.cos(a)).toFixed(2),
    y2: (C + r2 * Math.sin(a)).toFixed(2),
    major,
  };
});

function pad(n: number): string {
  return String(n).padStart(2, '0');
}

function clockParts(ts: number): { time: string; period: string } {
  const d = new Date(ts);
  const h = d.getHours();
  return {
    time: `${h % 12 || 12}:${pad(d.getMinutes())}`,
    period: h >= 12 ? 'م' : 'ص',
  };
}

  }, [phase, sessionIndex, totalSessions, studyMin, breakMin, toast]);

  const handlerRef = useRef(handlePhaseComplete);
  useEffect(() => {
    handlerRef.current = handlePhaseComplete;
  }, [handlePhaseComplete]);

  // ===== Timer interval =====
  useEffect(() => {
    if (status !== 'running' || deadline === null) return;
    let cancelled = false;
    const id = setInterval(() => {
      if (cancelled) return;
      const rem = Math.max(0, Math.ceil((deadline - Date.now()) / 1000));
      if (rem <= 0) {
        cancelled = true;
        clearInterval(id);
        handlerRef.current();
        return;
      }
      setRemaining(rem);
    }, 250);
    return () => {
      cancelled = true;
      clearInterval(id);
    };
  }, [status, deadline]);

  const handleStart = useCallback(() => {
    const alert = alertAudioRef.current;
    if (alert) {
      const v = alert.volume;
      alert.volume = 0;
      void alert
        .play()
        .then(() => {
          alert.pause();
          alert.currentTime = 0;
          alert.volume = v;
        })
        .catch(() => {
          alert.volume = v;
        });
    }

    setCelebrate(false);

    if (status === 'idle') {
      setPhase('study');
      setSessionIndex(0);
      setRemaining(studyMin * 60);
      setDeadline(Date.now() + studyMin * 60 * 1000);
      setStatus('running');
    } else if (status === 'paused') {
      setDeadline(Date.now() + remaining * 1000);
      setStatus('running');
    }
  }, [status, studyMin, remaining]);

    if (status !== 'running') return;
    setStatus('paused');
    setDeadline(null);
  }, [status]);

  const handleReset = useCallback(() => {
    setStatus('idle');
    setPhase('study');
    setSessionIndex(0);
    setRemaining(studyMin * 60);
    setDeadline(null);
    setCelebrate(false);
  }, [studyMin]);

    if (status === 'running') handlePause();
    else handleStart();
  }, [status, handlePause, handleStart]);

  useEffect(() => {

      if (e.metaKey || e.ctrlKey || e.altKey || e.repeat) return;
      const el = e.target as HTMLElement | null;
      const tag = el?.tagName;
      if (
        tag === 'INPUT' ||
        tag === 'TEXTAREA' ||
        tag === 'SELECT' ||
        el?.isContentEditable
      ) {
        return;
      }

      if (e.key === 'Escape' && focusMode) {
        setFocusMode(false);
        return;
      }
      if (tab !== 'timer') return;

      if (e.code === 'Space') {
        if (tag === 'BUTTON') return;
        e.preventDefault();
        handleToggle();
      } else if (e.code === 'KeyR') {
        if (status !== 'idle') handleReset();
      } else if (e.code === 'KeyF') {
        setFocusMode((v) => !v);
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [tab, focusMode, status, handleToggle, handleReset]);

  // ========================================================================
  // 🔊 Sound Handling (مع دعم الملفات الكبيرة)
  // ========================================================================
  /**
   * يوقف أي تحميل جاري، ويلغي المؤقّتات والـlisteners.
   * يُستدعى قبل تشغيل صوت جديد، أو عند unmount.

  const clearSoundLoad = useCallback(() => {

  }, []);

  const describeMediaError = useCallback((err: MediaError | null): string => {

    switch (err.code) {

        return 'تم إلغاء التحميل';
      case 2:
        return 'خطأ في الشبكة — تحقق من اتصالك';
      case 3:

      case 4:
        return 'صيغة الملف غير مدعومة في متصفحك';
      default:
        return 'تعذّر تشغيل الصوت';
    }
  }, []);

  const handleSoundSelect = useCallback(
    (id: string | null) => {
      const audio = ambientAudioRef.current;
      if (!audio) return;

      // أوقف أي صوت حالي + امسح أي تحميل معلّق
      clearSoundLoad();
      audio.pause();
      audio.removeAttribute('src');
      audio.load();

      // إلغاء الاختيار
      if (id === null || id === selectedSound) {
        setSelectedSound(null);
        setSoundLoading(null);
        return;
      }

      const sound = findSound(id);
      if (!sound) return;

      // ضع الصوت في حالة تحميل
      setSoundLoading(id);

      // ═══ معالجات الأحداث ═══
      const handleError = () => {
        const msg = describeMediaError(audio.error);
        console.error('[sound] error:', audio.error, sound.source);
        toast.show(`${msg}: ${sound.label}`, 'error');
        clearSoundLoad();
        audio.removeAttribute('src');
        audio.load();
        setSelectedSound(null);
        setSoundLoading(null);
      };

      const handleCanPlay = () => {
        // الصوت جاهز للتشغيل — شغّله
        clearSoundLoad();
        audio.loop = sound.loop;
        audio.volume = volume;
        void audio
          .play()
          .then(() => {
            setSelectedSound(id);
            setSoundLoading(null);
          })
          .catch((playErr: unknown) => {
            console.error('[sound] play failed:', playErr, sound.source);
            const msg =
              playErr instanceof DOMException && playErr.name === 'NotAllowedError'
                ? 'المتصفح منع التشغيل — اضغط على الصفحة أولاً ثم أعد المحاولة'
                : 'تعذّر تشغيل الصوت';
            toast.show(msg, 'error');
            setSelectedSound(null);
            setSoundLoading(null);
          });
      };

      audio.addEventListener('error', handleError);
      audio.addEventListener('canplay', handleCanPlay);

      // خزّن دالة التنظيف
      soundCleanupRef.current = () => {
        audio.removeEventListener('error', handleError);
        audio.removeEventListener('canplay', handleCanPlay);
      };

      // ═══ ابدأ التحميل ═══
      audio.preload = 'auto';
      audio.src = sound.source;
      audio.load();

      // ═══ Timeout احتياطي للملفات الضخمة ═══
      soundLoadTimeoutRef.current = setTimeout(() => {
        // إذا الصوت بدأ يشتغل، تجاهل
        if (!audio.paused && audio.readyState >= 2) return;

        toast.show(
          `انتهت مهلة التحميل لـ«${sound.label}» — الملف ضخم، جرّب مرة ثانية`,
          'error',
        );
        clearSoundLoad();
        audio.pause();
        audio.removeAttribute('src');
        audio.load();
        setSelectedSound(null);
        setSoundLoading(null);
      }, SOUND_LOAD_TIMEOUT_MS);
    },
    [selectedSound, volume, toast, clearSoundLoad, describeMediaError],
  );

    if (ambientAudioRef.current) {

    PRESETS.find(
      (p) =>
        p.study === studyMin &&
        p.breakMin === breakMin &&
        p.sessions === totalSessions,
    )?.id ?? null;

  const totalForPhase = phase === 'study' ? studyMin * 60 : breakMin * 60;
  const progress = totalForPhase > 0 ? 1 - remaining / totalForPhase : 0;
  const isBreakActive = phase === 'break' && status !== 'idle';
  const warning = status === 'running' && remaining <= 60 && remaining > 0;

  const emptyMessage =
    todayCount === 0
      ? 'لسا ما بدأت اليوم — جلسة واحدة تكفي لتفتح الطريق.'
      : `أنجزت ${todayCount} ${todayCount === 1 ? 'جلسة' : 'جلسات'} اليوم — استمر بهذا الإيقاع.`;

  const shellClass = focusMode
    ? 'campus-paper fixed inset-0 z-[60] overflow-y-auto'
    : 'campus-paper relative min-h-[calc(100dvh-var(--nav-h))] pb-24 md:pb-10';

  return (

      {/* ظلّ الراحة */}
      <div
        aria-hidden="true"
        className="pointer-events-none fixed inset-0 transition-opacity duration-[1600ms] motion-reduce:transition-none"
        style={{

            'radial-gradient(ellipse 80% 55% at 50% 35%, color-mix(in srgb, var(--gold) 9%, transparent), transparent 75%)',
        }}

      {/* خروج من الوضع الغامر */}
      {focusMode && (
        <button
          type="button"
          onClick={() => setFocusMode(false)}
          aria-label="الخروج من الوضع الغامر"
          className="fixed end-4 top-4 z-[70] flex h-10 w-10 items-center justify-center rounded-field border border-line bg-paper-soft text-ink-soft transition-colors hover:border-teal/40 hover:text-teal focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-teal"
        >

                {/* Presets */}
                <div
                  className="card-editorial ink-reveal p-5"
                  style={{ '--i': 1 } as CSSProperties}
                >
                  <h2 className="mb-3 flex items-center gap-2 text-sm font-bold text-ink">
                    <span aria-hidden="true">⏱️</span>

                  </h2>

                  <div className="grid grid-cols-3 gap-2">
                    {PRESETS.map((p) => {

                      return (
                        <button
                          key={p.id}
                          type="button"
                          onClick={() => {
                            setStudyMin(p.study);
                            setBreakMin(p.breakMin);
                            setTotalSessions(p.sessions);
                          }}
                          disabled={status === 'running'}
                          aria-pressed={active}
                          className={`rounded-field border px-2 py-3 text-center text-xs font-bold transition-all duration-200 motion-reduce:transition-none disabled:opacity-50 ${
                            active
                              ? 'border-teal bg-teal-tint text-teal'
                              : 'border-line bg-paper-soft text-ink-soft hover:border-teal/40'
                          }`}
                        >
                          <p className="num-inline font-mono text-sm leading-none">
                            {p.label}
                          </p>
                          <p className="mt-1 text-[10px] opacity-70">{p.hint}</p>
                        </button>
                      );
                    })}
                  </div>

                  <div className="mt-4 flex items-center justify-between gap-3 border-t border-line-soft pt-4">
                    <span className="text-xs font-bold text-ink-muted">
                      عدد الجلسات
                    </span>
                    <div className="flex items-center gap-2">
                      <button
                        type="button"
                        onClick={() =>
                          setTotalSessions((s) => Math.max(1, s - 1))
                        }
                        disabled={status === 'running' || totalSessions <= 1}
                        aria-label="تقليل"
                        className="flex h-8 w-8 items-center justify-center rounded-field border border-line bg-paper-soft text-lg font-bold text-ink-soft transition-colors hover:border-teal/40 hover:text-teal disabled:opacity-40"
                      >
                        −
                      </button>
                      <span className="num-inline min-w-[2ch] text-center font-mono text-base font-bold text-ink">
                        {totalSessions}
                      </span>
                      <button
                        type="button"
                        onClick={() =>
                          setTotalSessions((s) => Math.min(10, s + 1))
                        }
                        disabled={status === 'running' || totalSessions >= 10}
                        aria-label="زيادة"
                        className="flex h-8 w-8 items-center justify-center rounded-field border border-line bg-paper-soft text-lg font-bold text-ink-soft transition-colors hover:border-teal/40 hover:text-teal disabled:opacity-40"
                      >
                        +
                      </button>
                    </div>
                  </div>
                </div>

                {/* Sound */}
                <div
                  className="card-editorial ink-reveal p-5"
                  style={{ '--i': 2 } as CSSProperties}
                >
                  <SoundPicker
                    selectedId={selectedSound}
                    loadingId={soundLoading}
                    onSelect={handleSoundSelect}
                    volume={volume}
                    onVolumeChange={handleVolumeChange}
                  />
                </div>
              </aside>
            </div>
          </div>

          {/* ─── Health Tab ─── */}
          <div hidden={tab !== 'health'}>
            <HealthTab onGoToTimer={() => setTab('timer')} />
          </div>
        </div>
      </div>

      {/* Ambient audio element */}
      <audio ref={ambientAudioRef} preload="none" />
    </main>
  );
}

  /** يطلب من الصفحة تقريب منطقة (عند اختيار إعداد مسبق) */
  onZoomRequest?: (z: ZoomId) => void;
}

interface Palette { bone: string; muscle: string; context: string; sel: string; ok: string; bad: string; bg: string }

const URL_LITE = '/atlas/head-neck-lite.glb';

const LIGHT: Palette = { bone: '#E9E1CE', muscle: '#A8503A', context: '#C9AC9B', sel: '#0E6B73', ok: '#3D6E4A', bad: '#B8860B', bg: '#F7F6F1' };
const DARK: Palette = { bone: '#D2CAB5', muscle: '#D98B73', context: '#8F7B72', sel: '#7FD3DC', ok: '#7BC28A', bad: '#E8BC6A', bg: '#1D262B' };

function readPalette(el: HTMLElement): Palette {
  const cs = getComputedStyle(el);
  const v = (n: string): string => cs.getPropertyValue(n).trim();
  let dark = false;
  try {
    const deep = v('--paper-deep');
    dark = deep !== '' && new Color(deep).getHSL({ h: 0, s: 0, l: 0 }).l < 0.2;
  } catch {
    dark = false;
  }
  const fb = dark ? DARK : LIGHT;
  let context = fb.context;
  try {
    context = `#${new Color(fb.muscle).lerp(new Color(fb.bone), 0.55).getHexString()}`;
  } catch {
    /* القيمة الاحتياطية */
  }
  return { ...fb, context, ok: v('--success') || fb.ok, bg: v('--paper-soft') || fb.bg };
}

const hash01 = (s: string): number => {
  let h = 2166136261;
  for (let i = 0; i < s.length; i++) h = Math.imul(h ^ s.charCodeAt(i), 16777619);
  return ((h >>> 0) % 1000) / 1000;
};

interface Part extends PartInfo {
  label: string;
  geometry: BufferGeometry;
  center: Vector3;
  radius: number;
}

function usePartList(url: string): Part[] {
  const gltf = useGLTF(url);
  return useMemo(() => {
    const out: Part[] = [];
    gltf.scene.traverse((o: Object3D) => {
      const m = o as Mesh;
      if (!m.isMesh) return;
      const raw = (typeof m.userData.name === 'string' ? m.userData.name : m.name.replace(/_/g, ' ')) as string;
      const info = describePart(raw);
      m.geometry.computeBoundingBox();
      const box = m.geometry.boundingBox;
      const center = new Vector3();
      let radius = 0.2;
      if (box) {
        box.getCenter(center);
        radius = box.getSize(new Vector3()).length() / 2;
      }
      out.push({ ...info, label: labelOfPart(info, (id) => MUSCLE_BY_ID[id]?.en), geometry: m.geometry, center, radius });
    });
    return out;
  }, [gltf]);
}

const FRAMES: Record<ZoomId, { target: [number, number, number]; height: number }> = {
  whole: { target: [0, -1.1, 0.1], height: 6.3 },
  eyes: { target: [0, 0.4, 0.8], height: 1.4 },
  nose: { target: [0, 0.1, 0.9], height: 1.2 },
  mouth: { target: [0, -0.45, 0.8], height: 1.3 },
  side: { target: [0.6, 0.0, 0.3], height: 2.0 },
  neck: { target: [0, -1.5, 0.0], height: 3.4 },
  throat: { target: [0, -1.0, 0.4], height: 2.0 },
  posterior: { target: [0.7, -1.3, -0.1], height: 2.2 },
};

const VIEW_DIR: Record<ViewId, [number, number, number]> = {
  front: [0, 0.05, 1], left: [1, 0.05, 0], right: [-1, 0.05, 0], back: [0, 0.05, -1], top: [0, 1, 0.12], bottom: [0, -1, 0.2],
};

function reducedMotion(): boolean {
  try {
    return window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  } catch {
    return false;
  }
}

interface FocusReq { id: string; n: number }

function CameraRig({
  zoom, view, focus, parts, controls, freeTouch,
}: {
  zoom: ZoomId;
  view: { id: ViewId; n: number } | null;
  focus: FocusReq | null;
  parts: readonly Part[];
  controls: { current: ComponentRef<typeof OrbitControls> | null };
  freeTouch: boolean;
}) {
  const { camera, invalidate, size } = useThree();
  const aspect = size.width / Math.max(1, size.height);
  const goal = useRef<{ target: Vector3; dist: number; dir: Vector3 | null; instant: boolean } | null>(null);
  const first = useRef(true);
  const off = useMemo(() => new Vector3(), []);
  const lastDist = useRef(8);

  useEffect(() => {
    const el = (controls.current as unknown as { domElement?: HTMLElement } | null)?.domElement;
    if (el) el.style.touchAction = freeTouch ? 'none' : 'pan-y';
  }, [controls, freeTouch]);

  useEffect(() => {
    const f = FRAMES[zoom];
    const needW = Math.max(1.7, f.height * 0.62);
    const fit = Math.max(f.height, needW / aspect);
    const dist = fit / (2 * Math.tan(MathUtils.degToRad(FOV / 2)));
    lastDist.current = dist;
    goal.current = {
      target: new Vector3(...f.target),
      dist,
      dir: first.current ? new Vector3(...VIEW_DIR.front).normalize() : null,
      instant: first.current || reducedMotion(),
    };
    first.current = false;
    invalidate();
  }, [zoom, invalidate, aspect]);

  useEffect(() => {
    if (!view) return;
    const g = goal.current;
    goal.current = {
      target: g?.target ?? new Vector3(...FRAMES[zoom].target),
      dist: g?.dist ?? lastDist.current,
      dir: new Vector3(...VIEW_DIR[view.id]).normalize(),
      instant: reducedMotion(),
    };
    invalidate();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [view, invalidate]);

  // الانتقال إلى العضلة المحددة (الجهة الأقرب للكاميرا)
  useEffect(() => {
    if (!focus) return;
    const cands = parts.filter((p) => p.muscleId === focus.id);
    if (cands.length === 0) return;
    let best = cands[0];
    let bd = Infinity;
    for (const p of cands) {
      const d = p.center.distanceTo(camera.position);
      if (d < bd) { bd = d; best = p; }
    }
    if (!best) return;
    const radius = cands.filter((p) => Math.sign(p.center.x) === Math.sign(best.center.x)).reduce((m, p) => Math.max(m, p.radius), best.radius);
    const dist = MathUtils.clamp(radius * 3.4, 1.3, 5);
    lastDist.current = dist;
    goal.current = { target: best.center.clone(), dist, dir: null, instant: reducedMotion() };
    invalidate();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [focus?.n]);

  useFrame((_, dt) => {
    const c = controls.current;
    const g = goal.current;
    if (!c || !g) return;
    const k = g.instant ? 1 : 1 - Math.exp(-dt * 7);
    off.copy(camera.position).sub(c.target);
    let len = off.length();
    if (len < 1e-4) off.set(0, 0.05, 1);
    if (g.dir) off.normalize().lerp(g.dir, k).normalize();
    else off.normalize();
    len = MathUtils.lerp(len, g.dist, k);
    off.multiplyScalar(len);
    c.target.lerp(g.target, k);
    camera.position.copy(c.target).add(off);
    c.update();
    const dirDone = !g.dir || off.clone().normalize().distanceTo(g.dir) < 0.01;
    if (c.target.distanceTo(g.target) < 0.002 && Math.abs(len - g.dist) < 0.01 && dirDone) goal.current = null;
    else invalidate();
  });

  return null;
}

// ==================== حركة الرأس والفك ====================
const HEAD_PIVOT: [number, number, number] = [0, -0.09, -0.12];
const JAW_PIVOT: [number, number, number] = [0, 0.18, 0.14];
const neg = (p: [number, number, number]): [number, number, number] => [-p[0], -p[1], -p[2]];

function HeadRig({ pose, kRef, children }: { pose: Pose; kRef: { current: number }; children: ReactNode }) {
  const ref = useRef<Group>(null);
  const invalidate = useThree((s) => s.invalidate);
  const animated = (pose.tilt ?? 0) !== 0 || (pose.turn ?? 0) !== 0 || (pose.fx ?? 0) !== 0;

  useFrame((_, dt) => {
    const g = ref.current;
    if (!g) return;
    const k = kRef.current;
    const tx = (pose.fx ?? 0) * k * 0.28;
    const ty = (pose.turn ?? 0) * k * 0.45;
    const tz = -(pose.tilt ?? 0) * k * 0.25;
    g.rotation.x = MathUtils.damp(g.rotation.x, tx, 14, dt);
    g.rotation.y = MathUtils.damp(g.rotation.y, ty, 14, dt);
    g.rotation.z = MathUtils.damp(g.rotation.z, tz, 14, dt);
    const moving = Math.abs(g.rotation.x - tx) > 5e-4 || Math.abs(g.rotation.y - ty) > 5e-4 || Math.abs(g.rotation.z - tz) > 5e-4;
    if (animated || moving) invalidate();
  });

  return (
    <group ref={ref} position={HEAD_PIVOT}>
      <group position={neg(HEAD_PIVOT)}>{children}</group>
    </group>
  );
}

function JawRig({ pose, kRef, children }: { pose: Pose; kRef: { current: number }; children: ReactNode }) {
  const ref = useRef<Group>(null);
  const invalidate = useThree((s) => s.invalidate);
  const animated = (pose.jaw ?? 0) !== 0;

  useFrame((_, dt) => {
    const g = ref.current;
    if (!g) return;
    const target = Math.max(0, pose.jaw ?? 0) * kRef.current * 0.2;
    g.rotation.x = MathUtils.damp(g.rotation.x, target, 14, dt);
    if (animated || Math.abs(g.rotation.x - target) > 5e-4) invalidate();
  });

  return (
    <group ref={ref} position={JAW_PIVOT}>
      <group position={neg(JAW_PIVOT)}>{children}</group>
    </group>
  );
}

interface ShaderLike { fragmentShader: string; uniforms: Record<string, { value: unknown }> }

/** إضاءة حافّة خفيفة (Fresnel) تُبرز شكل العضلة وحجمها */
function rimPatch(shader: ShaderLike): void {
  shader.uniforms.rimColor = { value: new Color('#ffffff') };
  shader.fragmentShader = shader.fragmentShader
    .replace('#include <common>', '#include <common>\nuniform vec3 rimColor;')
    .replace(
      '#include <emissivemap_fragment>',
      '#include <emissivemap_fragment>\n  float rimF = pow(1.0 - saturate(dot(normalize(normal), normalize(vViewPosition))), 2.6);\n  totalEmissiveRadiance += rimColor * rimF * 0.2;',
    );
}

interface Look {
  palette: Palette;
  tones: ReadonlyMap<string, Tone>;
  highlight: ReadonlySet<string> | null;
  vis: Visibility;
  autoHide: ReadonlySet<string>;
  peel: Depth;
  boneOpacity: number;
  muscleOpacity: number;
  focus: boolean;
  anySelected: boolean;
  moving: boolean;
  plane: Plane | null;
  kRef: { current: number };
  hover: string | null;
  pickedKey: string | null;
  setHover: (key: string | null, label?: string) => void;
  onPickPart: (p: Part) => void;
}

function PartMesh({ part, look }: { part: Part; look: Look }) {
  const gl = useThree((s) => s.gl);
  const invalidate = useThree((s) => s.invalidate);
  const mat = useRef<MeshStandardMaterial>(null);
  const { palette, tones, highlight } = look;
  const id = part.muscleId;
  const tone = id ? tones.get(id) : look.pickedKey === part.key ? ('sel' as const) : undefined;
  const pulse = tone === 'sel' && look.moving && id !== null;

  useFrame(() => {
    const m = mat.current;
    if (!m || !pulse) return;
    m.emissiveIntensity = 0.1 + 0.55 * look.kRef.current;
    invalidate();
  });

  const baseColor = useMemo(() => {
    const base = part.kind === 'bone' ? palette.bone : id ? palette.muscle : palette.context;
    const c = new Color(base);
    c.offsetHSL(0, 0, (hash01(part.name) - 0.5) * (part.kind === 'bone' ? 0.05 : 0.11));
    return `#${c.getHexString()}`;
  }, [part.kind, part.name, id, palette]);

  const hidden =
    !partVisible(part, look.vis, look.peel) || (id !== null && look.autoHide.has(id) && !tone);
  if (hidden) return null;

  const dim = id !== null && highlight !== null && !highlight.has(id) && !tone;
  const faded = look.focus && look.anySelected && part.kind === 'muscle' && !tone;
  const hovered = look.hover === part.key;
  let color = baseColor;
  if (tone === 'sel') color = palette.sel;
  else if (tone === 'ok') color = palette.ok;
  else if (tone === 'bad') color = palette.bad;
  else if (hovered) color = `#${new Color(baseColor).lerp(new Color('#ffffff'), 0.25).getHexString()}`;

  const base = part.kind === 'bone' ? look.boneOpacity : look.muscleOpacity;
  const opacity = dim ? Math.min(base, 0.12) : faded ? Math.min(base, 0.3) : base;
  const transparent = opacity < 1;
  const sectioned = look.plane !== null;

  return (
    <mesh
      key={sectioned ? 's' : 'n'}
      geometry={part.geometry}
      renderOrder={transparent ? 1 : 0}
      onClick={(e: ThreeEvent<MouseEvent>) => {
        e.stopPropagation();
        if (e.delta > 6) return;
        look.onPickPart(part);
      }}
      onPointerOver={(e: ThreeEvent<PointerEvent>) => {
        e.stopPropagation();
        look.setHover(part.key, part.label);
        gl.domElement.style.cursor = 'pointer';
      }}
      onPointerOut={() => {
        look.setHover(null);
        gl.domElement.style.cursor = '';
      }}
    >
      <meshStandardMaterial
        ref={mat}
        color={color}
        emissive={tone || hovered ? color : '#000000'}
        emissiveIntensity={tone === 'sel' ? 0.22 : tone ? 0.18 : hovered ? 0.1 : 0}
        roughness={part.kind === 'bone' ? 0.8 : 0.46}
        metalness={0.02}
        transparent={transparent}
        opacity={opacity}
        depthWrite={!transparent}
        side={sectioned ? 2 : 0}
        clippingPlanes={look.plane ? [look.plane] : null}
        onBeforeCompile={rimPatch}
      />
    </mesh>
  );
}

function PinnedLabel({ id, parts }: { id: string; parts: readonly Part[] }) {

  const cands = useMemo(() => parts.filter((p) => p.muscleId === id), [parts, id]);
  const camera = useThree((s) => s.camera);

  useFrame(() => {
    const g = ref.current;
    if (!g || cands.length === 0) return;
    let best = cands[0];
    let bd = Infinity;
    for (const p of cands) {
      const d = p.center.distanceTo(camera.position);
      if (d < bd) { bd = d; best = p; }
    }
    if (best) g.position.copy(best.center);
  });

  if (cands.length === 0) return null;

      <Html center zIndexRange={[5, 0]} style={{ pointerEvents: 'none' }}>
        <span className="fa3-pin" dir="ltr">{m?.short ?? m?.en ?? id}</span>
      </Html>
    </group>
  );
}

interface SceneProps extends Atlas3DProps {
  url: string;
  palette: Palette;
  ts: ToolState;
  view: { id: ViewId; n: number } | null;
  focusReq: FocusReq | null;
  autoHide: ReadonlySet<string>;
  hover: string | null;
  pickedKey: string | null;
  setHover: (key: string | null, label?: string) => void;
  onPickPart: (p: Part) => void;
  partsRef: { current: readonly Part[] };
  captureRef: { current: (() => string | null) | null };
  bg: string;
}

function Scene(props: SceneProps) {
  const { ts } = props;
  const parts = usePartList(props.url);
  const controls = useRef<ComponentRef<typeof OrbitControls>>(null);
  const { gl, scene, camera, invalidate } = useThree();

  useEffect(() => {
    props.partsRef.current = parts;
  }, [parts, props.partsRef]);

  useEffect(() => {
    gl.localClippingEnabled = true;
  }, [gl]);

  useEffect(() => {
    props.captureRef.current = () => {
      const prev = gl.getClearAlpha();
      gl.setClearColor(new Color(props.bg), 1);
      gl.render(scene as ThreeScene, camera);
      const url = gl.domElement.toDataURL('image/png');
      gl.setClearColor(new Color(props.bg), prev);
      invalidate();
      return url;
    };
    return () => {
      props.captureRef.current = null;
    };
  }, [gl, scene, camera, invalidate, props.bg, props.captureRef]);

  useFrame(() => {
    if (ts.autoRotate) invalidate();
  });

  const plane = useMemo(() => new Plane(new Vector3(-1, 0, 0), 0), []);
  if (ts.section.on) {
    const n = ts.section.axis === 'x' ? [1, 0, 0] : ts.section.axis === 'y' ? [0, 1, 0] : [0, 0, 1];
    const s = ts.section.flip ? 1 : -1;

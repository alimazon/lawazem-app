    plane.normal.set(n[0] ?? 0, n[1] ?? 0, n[2] ?? 0).multiplyScalar(s);
    plane.constant = ts.section.flip ? -ts.section.pos : ts.section.pos;
  }

  const tones = useMemo(() => new Map(props.selItems.map((s) => [s.id, s.tone] as const)), [props.selItems]);
  const moving = Object.keys(props.pose).length > 0;
  const look: Look = {
    palette: props.palette,
    tones,
    highlight: props.highlight,
    vis: ts.vis,
    autoHide: props.autoHide,
    peel: ts.peel,
    boneOpacity: ts.boneOpacity,
    muscleOpacity: ts.muscleOpacity,
    focus: ts.focus && !props.hideNames,
    anySelected: props.selItems.length > 0,
    moving,
    plane: ts.section.on ? plane : null,
    kRef: props.kRef,
    hover: props.hover,
    pickedKey: props.pickedKey,
    setHover: props.setHover,
    onPickPart: props.onPickPart,
  };

  const fixed = parts.filter((p) => !p.head);
  const head = parts.filter((p) => p.head && !p.jaw);
  const jaw = parts.filter((p) => p.head && p.jaw);
  const selIds = props.selItems.filter((s) => s.tone === 'sel').map((s) => s.id);

  return (
    <>
      <ambientLight intensity={0.62} />
      <hemisphereLight args={['#ffffff', '#b8a99a', 0.9]} />
      <directionalLight position={[3, 4, 6]} intensity={2.0} />
      <directionalLight position={[-4, 1, 3]} intensity={0.8} />
      <directionalLight position={[0, 3, -5]} intensity={0.7} />

      <OrbitControls
        ref={controls}
        makeDefault
        enableDamping
        dampingFactor={0.12}
        rotateSpeed={0.7}
        zoomSpeed={0.8}
        autoRotate={ts.autoRotate}
        autoRotateSpeed={1.4}
        screenSpacePanning
        minDistance={0.7}
        maxDistance={22}
        touches={{ ONE: TOUCH.ROTATE, TWO: TOUCH.DOLLY_PAN }}
      />
      <CameraRig zoom={props.zoom} view={props.view} focus={props.focusReq} parts={parts} controls={controls} freeTouch={ts.freeTouch} />

      <group>{fixed.map((p) => <PartMesh key={p.key} part={p} look={look} />)}</group>
      <HeadRig pose={props.pose} kRef={props.kRef}>
        {head.map((p) => <PartMesh key={p.key} part={p} look={look} />)}
        <JawRig pose={props.pose} kRef={props.kRef}>
          {jaw.map((p) => <PartMesh key={p.key} part={p} look={look} />)}
        </JawRig>
      </HeadRig>

      {ts.labels && !props.hideNames && selIds.length <= 3 && <PinnedLabels ids={selIds} parts={parts} />}
    </>
  );
}

const INITIAL_TS = (): ToolState => {
  let quality: 'full' | 'lite' = 'full';
  let freeTouch = false;
  try {
    const stored = localStorage.getItem(QUALITY_KEY);
    const mem = (navigator as Navigator & { deviceMemory?: number }).deviceMemory ?? 8;
    const small = Math.min(window.screen.width, window.screen.height) < 700;
    quality = stored === 'lite' || stored === 'full' ? stored : small || mem <= 2 ? 'lite' : 'full';
    freeTouch = window.innerWidth >= 700;
  } catch {
    /* القيم الافتراضية */
  }
  return {
    vis: DEFAULT_VISIBILITY,
    peel: 1,
    boneOpacity: 1,
    muscleOpacity: 1,
    section: { on: false, axis: 'x', pos: 0, flip: false } satisfies Section,
    autoRotate: false,
    labels: true,
    focus: true,
    freeTouch,
    quality,
  };
};

interface BoundaryProps {
  onFail: () => void;
  children: ReactNode;
}

/** يلتقط أي خطأ في الـ 3D (فشل تحميل الحزمة أو WebGL) ويعيد المستخدم إلى الرسم 2D */
class Boundary extends Component<BoundaryProps, { failed: boolean }> {
  state = { failed: false };

  static getDerivedStateFromError(): { failed: boolean } {
    return { failed: true };
  }

  componentDidCatch(): void {
    this.props.onFail();
  }

  render(): ReactNode {
    return this.state.failed ? null : this.props.children;
  }
}

  return (
    <Boundary onFail={props.onFail}>
      <Atlas3D {...props} />
    </Boundary>
  );

function boneGroup(name: string): string {
  if (/mandible/.test(name)) return 'b:mandible';
  if (/hyoid/.test(name)) return 'b:hyoid';
  if (/^(atlas|axis)$|cervical vertebra/.test(name)) return 'b:cervical';
  if (/thoracic vertebra/.test(name)) return 'b:thoracic';
  if (/rib$/.test(name)) return 'b:ribs';
  if (/sternum|manubrium|xiphoid/.test(name)) return 'b:sternum';
  if (/clavicle/.test(name)) return 'b:clavicle';
  if (/scapula/.test(name)) return 'b:scapula';
  return 'b:cranium';
}

function contextGroup(name: string): string {
  if (/pectoralis|deltoid|subclavius|serratus|ascending part/.test(name)) return 'c:chest';
  if (/pterygoid/.test(name)) return 'c:pterygoid';
  if (/capitis posterior|longissimus|splenius cervicis/.test(name)) return 'c:suboccipital';
  return 'c:eye';
}

export interface PartInfo {

  kind: 'bone' | 'muscle';
  name: string;
  muscleId: string | null;

  head: boolean;

  jaw: boolean;

const JAW_RE = /^(mandible|(left|right) (superficial part of masseter|deep part of masseter|medial pterygoid|inferior head of lateral pterygoid|mentalis|depressor labii inferioris|depressor anguli oris))$/;
const HEAD_CONTEXT_RE = /((superior|inferior|medial|lateral) rectus|superior oblique|inferior oblique|levator palpebrae|pterygoid)/;

const titleCase = (s: string): string => s.charAt(0).toUpperCase() + s.slice(1);

export function describePart(raw: string): PartInfo {
  const kind = raw.startsWith('b|') ? 'bone' : 'muscle';
  const name = raw.slice(2);
  const muscleId = kind === 'muscle' ? (OWNER.get(name) ?? null) : null;
  const group =
    kind === 'bone' ? boneGroup(name) : muscleId ? (MUSCLE_GROUP_OF.get(muscleId) ?? 'c:eye') : contextGroup(name);
  const head =
    kind === 'bone'
      ? HEAD_BONE_SET.has(name)
      : muscleId
        ? !NECK_GROUP_MEMBER.has(muscleId)
        : HEAD_CONTEXT_RE.test(name) && !/capitis/.test(name);
  const depth: Depth = muscleId ? (MUSCLE_DEPTH[muscleId] ?? 2) : contextDepth(name);
  return { key: raw, kind, name, muscleId, group, head, jaw: JAW_RE.test(name) || /pterygoid/.test(name), depth };
}

const NECK_GROUP_MEMBER = new Set<string>(
  MUSCLE_GROUPS.filter((g) => NECK_SET.has(g.labelEn)).flatMap((g) => [...g.ids]),
);

export function labelOfPart(p: PartInfo, en: (id: string) => string | undefined): string {
  const side = /^left /.test(p.name) ? 'Left' : /^right /.test(p.name) ? 'Right' : '';
  const base = p.muscleId ? (en(p.muscleId) ?? p.name) : titleCase(p.name.replace(/^(left|right) /, '').replace(/ \(2\)$/, ''));
  return side ? `${base} · ${side}` : base;
}

  if (v.groups.has(p.group) || v.parts.has(p.key)) return false;
  if (p.kind === 'muscle') {
    if (p.muscleId && v.muscles.has(p.muscleId)) return false;
    if (p.depth < peel) return false;
  }

}

  };

        muscles: new Set(),
        parts: new Set(),
      };

    case 'posterior':
      return { ...only(POSTERIOR_KEEP, []), groups: new Set(['c:eye', 'c:pterygoid', 'c:chest']) };
    case 'skeleton':

/** نافذة حوارية بسيطة: Esc للإغلاق، حبس التركيز، وإعادته للعنصر السابق، وقفل تمرير الصفحة. */
import { useEffect, useRef, type ReactNode } from 'react';

const FOCUSABLE = 'button:not([disabled]), [href], input:not([disabled]), select, textarea, [tabindex]:not([tabindex="-1"])';

export default function AtlasDialog({
  open, onClose, title, wide, children,
}: { open: boolean; onClose: () => void; title: string; wide?: boolean; children: ReactNode }) {
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    if (!open) return;
    const prev = document.activeElement as HTMLElement | null;
    const overflow = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    ref.current?.focus();

    const onKey = (e: KeyboardEvent): void => {
      if (e.key === 'Escape') {
        e.stopPropagation();
        onClose();
        return;
      }
      if (e.key !== 'Tab' || !ref.current) return;
      const els = [...ref.current.querySelectorAll<HTMLElement>(FOCUSABLE)];
      const first = els[0];
      const last = els[els.length - 1];
      if (!first || !last) return;
      if (e.shiftKey && document.activeElement === first) {
        e.preventDefault();
        last.focus();
      } else if (!e.shiftKey && document.activeElement === last) {
        e.preventDefault();
        first.focus();
      }
    };
    document.addEventListener('keydown', onKey, true);
    return () => {
      document.removeEventListener('keydown', onKey, true);
      document.body.style.overflow = overflow;
      prev?.focus?.();
    };
  }, [open, onClose]);

  if (!open) return null;
  return (
    <div className="fa-modal-overlay" onClick={onClose}>
      <div
        ref={ref}
        className={`fa-modal${wide ? ' is-wide' : ''}`}
        role="dialog"
        aria-modal="true"
        aria-label={title}
        tabIndex={-1}
        dir="rtl"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="fa-modal-head">
          <h2>{title}</h2>
          <button type="button" className="fa-btn" onClick={onClose}>إغلاق</button>
        </div>
        <div className="fa-modal-body">{children}</div>
      </div>
    </div>
  );
}

  const c = t % CYCLE;
  if (c < 800) return ease(c / 800);
  if (c < 1900) return 1;
  if (c < 2600) return 1 - ease((c - 1900) / 700);
  return 0;
}

export type Innervation = 'VII' | 'V3';

export function innervationOf(m: Muscle): Innervation | null {
  const vii = /\bCN VII\b/.test(m.nerve);
  const v3 = /\bCN V3\b/.test(m.nerve);
  if (vii && v3) return null; // عضلة مختلطة (مثل digastric)
  if (vii) return 'VII';
  if (v3) return 'V3';
  return null;
}

const NERVE_TAGS: readonly (readonly [RegExp, string])[] = [
  [/\bCN VII\b/, 'CN VII'],
  [/\bCN V3\b/, 'CN V3'],
  [/\bCN XI\b/, 'CN XI'],
  [/\bCN XII\b/, 'CN XII'],
  [/ansa cervicalis/i, 'Ansa cervicalis'],
  [/dorsal scapular/i, 'Dorsal scapular n.'],
  [/anterior rami/i, 'Anterior rami'],
  [/posterior rami/i, 'Posterior rami'],
];

/** وسوم العصب التي تظهر كشارات على البطاقة. */
export function nerveTagsOf(m: Muscle): string[] {

}

export function pharyngealArchOf(m: Muscle): 1 | 2 | null {
  const inn = innervationOf(m);
  return inn === 'VII' ? 2 : inn === 'V3' ? 1 : null;
}

export function validateAtlas(): string[] {
  const problems: string[] = [];
  const ids = new Set<string>();

  for (const m of MUSCLES) {
    if (ids.has(m.id)) problems.push(`duplicate id: ${m.id}`);
    ids.add(m.id);

    if (!m.nerve.trim()) problems.push(`${m.id}: missing nerve`);

    const g = m.geometry;
    const fans = g.type === 'fan' ? [g] : g.type === 'multi' ? g.parts : [];
    for (const f of fans) {
      if (f.A.length !== 8 || f.B.length !== 8) {
        problems.push(`${m.id}: fan geometry needs 8 numbers in A and B`);
      }
    }
    if (g.type === 'multi' && g.parts.length < 2) problems.push(`${m.id}: multi geometry needs 2 or more parts`);
    if (g.type === 'ring' && (g.irx >= g.rx || g.iry >= g.ry)) {
      problems.push(`${m.id}: ring inner radius must be smaller than outer radius`);
    }
  }

  const seen = new Map<string, number>();
  for (const g of MUSCLE_GROUPS) {
    for (const id of g.ids) {
      if (!ids.has(id)) problems.push(`group "${g.labelEn}" references unknown muscle: ${id}`);
      seen.set(id, (seen.get(id) ?? 0) + 1);
    }
  }
  for (const id of ids) {
    const c = seen.get(id) ?? 0;
    if (c === 0) problems.push(`${id}: not in any group`);
    if (c > 1) problems.push(`${id}: appears in ${c} groups`);
  }

  return problems;
}

if (process.env.NODE_ENV !== 'production') {
  const issues = validateAtlas();
  if (issues.length) console.warn('[anatomy] data issues:\n' + issues.join('\n'));
}

// lib/anatomy-geometry.ts
import { MUSCLES, type FanGeometry, type Muscle, type MultiGeometry, type RingGeometry } from './anatomy-data';

export interface BuiltGeometry {
  d: string;
  dark: string;
  light: string;
  evenodd: boolean;
}

const f1 = (n: number): number => Math.round(n * 10) / 10;
const at = (a: readonly number[], i: number): number => a[i] ?? 0;

function fan(g: FanGeometry): BuiltGeometry {
  const { A, B, n, bulge } = g;
  let dark = '';
  let light = '';

  for (let i = 1; i < n; i++) {
    const u = i / n;
    const p: number[] = [];
    for (let k = 0; k < 8; k++) p.push(f1(at(A, k) + (at(B, k) - at(A, k)) * u));
    const seg = `M${p[0]},${p[1]}C${p[2]},${p[3]} ${p[4]},${p[5]} ${p[6]},${p[7]}`;
    if (i % 2) dark += seg;
    else light += seg;
  }

  const cap = (
    px: number, py: number, qx: number, qy: number,
    pcx: number, pcy: number, qcx: number, qcy: number,
  ): string => {
    const mx = (px + qx) / 2;
    const my = (py + qy) / 2;
    const cx = (pcx + qcx) / 2;
    const cy = (pcy + qcy) / 2;
    const dx = mx - cx;
    const dy = my - cy;
    const len = Math.hypot(dx, dy) || 1;
    const w = Math.hypot(px - qx, py - qy);
    const off = w < 2 ? 0 : 2 * bulge;
    return `${f1(mx + (dx / len) * off)},${f1(my + (dy / len) * off)}`;
  };

  const end = cap(at(A, 6), at(A, 7), at(B, 6), at(B, 7), at(A, 4), at(A, 5), at(B, 4), at(B, 5));
  const start = cap(at(B, 0), at(B, 1), at(A, 0), at(A, 1), at(B, 2), at(B, 3), at(A, 2), at(A, 3));

  const d =
    `M${A[0]},${A[1]}C${A[2]},${A[3]} ${A[4]},${A[5]} ${A[6]},${A[7]}` +
    `Q${end} ${B[6]},${B[7]}` +
    `C${B[4]},${B[5]} ${B[2]},${B[3]} ${B[0]},${B[1]}` +
    `Q${start} ${A[0]},${A[1]}Z`;

  return { d, dark, light, evenodd: false };
}

function ell(cx: number, cy: number, rx: number, ry: number): string {
  return (
    `M${f1(cx - rx)},${f1(cy)}` +
    `a${f1(rx)},${f1(ry)} 0 1,0 ${f1(2 * rx)},0` +
    `a${f1(rx)},${f1(ry)} 0 1,0 ${f1(-2 * rx)},0Z`
  );
}

function ring(g: RingGeometry): BuiltGeometry {
  let dark = '';
  let light = '';
  for (let i = 1; i < g.n; i++) {
    const u = i / g.n;
    const p = ell(g.cx, g.cy, g.irx + (g.rx - g.irx) * u, g.iry + (g.ry - g.iry) * u);
    if (i % 2) dark += p;
    else light += p;
  }
  return {
    d: ell(g.cx, g.cy, g.rx, g.ry) + ell(g.cx, g.cy, g.irx, g.iry),
    dark,
    light,
    evenodd: true,
  };
}

function multi(g: MultiGeometry): BuiltGeometry {
  const built = g.parts.map(fan);
  return {
    d: built.map((b) => b.d).join(''),
    dark: built.map((b) => b.dark).join(''),
    light: built.map((b) => b.light).join(''),
    evenodd: false,
  };
}

export function buildGeometry(m: Muscle): BuiltGeometry {
  const g = m.geometry;
  if (g.type === 'ring') return ring(g);
  if (g.type === 'multi') return multi(g);
  return fan(g);
}

export const GEOMETRY: Readonly<Record<string, BuiltGeometry>> = (() => {
  const out: Record<string, BuiltGeometry> = {};
  for (const m of MUSCLES) out[m.id] = buildGeometry(m);
  return out;
})();

/** حالة التعلّم الموحّدة: التقدّم، المفضّلة، الملاحظات، وبطاقات المراجعة المتباعدة. تُحفَظ محلياً على الجهاز. */
import { MUSCLES, MUSCLE_BY_ID } from './anatomy-data';
import { grade as srsGrade, isDue, newCard, type Grade, type SrsState } from './anatomy-srs';

export interface MuscleProgress {
  seen: number;
  correct: number;
  wrong: number;
  /** مستوى الإتقان 0..5 */
  mastery: number;
  last: number;
}

export interface LearnState {
  progress: Record<string, MuscleProgress>;
  favorites: string[];
  notes: Record<string, string>;
  srs: SrsState;
}

export const LEARN_KEY = 'anatomy-learn';
export const LEARN_VERSION = 2;

export const DEFAULT_LEARN: LearnState = { progress: {}, favorites: [], notes: {}, srs: { cards: {} } };

const EMPTY: MuscleProgress = { seen: 0, correct: 0, wrong: 0, mastery: 0, last: 0 };

export function markSeen(s: LearnState, id: string): LearnState {
  if (!MUSCLE_BY_ID[id]) return s;
  const p = s.progress[id] ?? EMPTY;
  return { ...s, progress: { ...s.progress, [id]: { ...p, seen: p.seen + 1, last: Date.now() } } };
}

function masteryOf(correct: number, wrong: number): number {
  const total = correct + wrong;
  if (total === 0) return 0;
  return Math.round((correct / total) * Math.min(1, total / 5) * 5);
}

/**
 * يسجّل إجابة ويقيّم بطاقة المراجعة مرة واحدة فقط.
 * `g` اختياري: من البطاقات يُمرَّر تقييم المستخدم، ومن الاختبار تُستخدم درجة افتراضية (4 صحيح، 1 خطأ).
 */
export function recordAnswer(s: LearnState, id: string, correct: boolean, g?: Grade): LearnState {
  if (!MUSCLE_BY_ID[id]) return s;
  const p = s.progress[id] ?? EMPTY;
  const c = p.correct + (correct ? 1 : 0);
  const w = p.wrong + (correct ? 0 : 1);
  const progress: MuscleProgress = { ...p, correct: c, wrong: w, mastery: masteryOf(c, w), last: Date.now() };
  const card = s.srs.cards[id] ?? newCard(id);
  const updated = srsGrade(card, g ?? (correct ? 4 : 1));
  return { ...s, progress: { ...s.progress, [id]: progress }, srs: { cards: { ...s.srs.cards, [id]: updated } } };
}

export function toggleFavorite(s: LearnState, id: string): LearnState {
  const has = s.favorites.includes(id);
  return { ...s, favorites: has ? s.favorites.filter((x) => x !== id) : [...s.favorites, id] };
}

export function setNote(s: LearnState, id: string, note: string): LearnState {
  const notes = { ...s.notes };
  const t = note.trim();
  if (t) notes[id] = t;
  else delete notes[id];
  return { ...s, notes };
}

export interface Stats {
  total: number;
  seen: number;
  mastered: number;
  accuracy: number;
  due: number;
  favorites: number;
  notes: number;
  perRegion: Record<string, { total: number; mastered: number }>;
}

export function computeStats(s: LearnState): Stats {
  let seen = 0;
  let mastered = 0;
  let correct = 0;
  let wrong = 0;
  const perRegion: Stats['perRegion'] = {};
  for (const m of MUSCLES) {
    const region = m.region ?? 'Other';
    const r = (perRegion[region] ??= { total: 0, mastered: 0 });
    r.total++;
    const p = s.progress[m.id];
    if (!p) continue;
    if (p.seen > 0 || p.correct + p.wrong > 0) seen++;
    if (p.mastery >= 4) {
      mastered++;
      r.mastered++;
    }
    correct += p.correct;
    wrong += p.wrong;
  }
  return {
    total: MUSCLES.length,
    seen,
    mastered,
    accuracy: correct + wrong > 0 ? correct / (correct + wrong) : 0,
    due: dueForReview(s).length,
    favorites: s.favorites.length,
    notes: Object.keys(s.notes).length,
    perRegion,
  };
}

/** بطاقات استحقّت مراجعتها (الأقدم أولاً) */
export function dueForReview(s: LearnState): string[] {
  return Object.values(s.srs.cards)
    .filter((c) => MUSCLE_BY_ID[c.id] && isDue(c))
    .sort((a, b) => a.due - b.due)
    .map((c) => c.id);
}

/** عضلات لم تُراجَع بعد (لبدء الدراسة قبل أي اختبار) */
export function newForStudy(s: LearnState, limit: number): string[] {
  return MUSCLES.filter((m) => !s.srs.cards[m.id]).map((m) => m.id).slice(0, limit);
}

export function buildStudyQueue(s: LearnState, size = 20): string[] {
  const due = dueForReview(s).slice(0, size);
  return [...due, ...newForStudy(s, Math.max(0, Math.min(10, size - due.length)))];
}

export function exportLearn(s: LearnState): string {
  return JSON.stringify({ version: LEARN_VERSION, exportedAt: Date.now(), data: s }, null, 2);
}

export function importLearn(json: string): LearnState | null {
  try {
    const parsed = JSON.parse(json) as { data?: Partial<LearnState> };
    const d = parsed.data;
    if (!d || typeof d !== 'object') return null;
    return {
      progress: d.progress ?? {},
      favorites: Array.isArray(d.favorites) ? d.favorites.filter((id) => MUSCLE_BY_ID[id]) : [],
      notes: d.notes ?? {},
      srs: d.srs ?? { cards: {} },
    };
  } catch {
    return null;
  }
}

// lib/anatomy-pose.ts
export type Pose = Readonly<Record<string, number>>;

const r2 = (x: number): number => Math.round(x * 100) / 100;
const clamp01 = (x: number): number => Math.max(0, Math.min(1, x));

export function scalePose(pose: Pose, k: number): Pose {
  const out: Record<string, number> = {};
  for (const key of Object.keys(pose)) out[key] = (pose[key] ?? 0) * k;
  return out;
}

export function combinePoses(
  parts: readonly { pose: Pose; w?: number }[],
  extra: Pose = {},
): Pose {
  const out: Record<string, number> = {};
  for (const { pose, w = 1 } of parts) {
    for (const key of Object.keys(pose)) {
      const v = (pose[key] ?? 0) * w;
      if (Math.abs(v) > Math.abs(out[key] ?? 0)) out[key] = v;
    }
  }
  for (const key of Object.keys(extra)) out[key] = extra[key] ?? 0;
  return out;
}

export function computeFace(P: Pose): FaceShape {
  const g = (k: string): number => P[k] ?? 0;

  const hy = r2(222 + g('jaw') * 10);
  const head =
    `M110,14 C62,14 40,52 40,100 C40,142 50,176 68,198 C80,212 96,${hy} 110,${hy} ` +
    `C124,${hy} 140,212 152,198 C170,176 180,142 180,100 C180,52 158,14 110,14 Z`;

  const e = g('eye');
  const eyePath = (cx: number): string =>
    `M${cx - 14},98 Q${cx},${r2(82 + 22 * e)} ${cx + 14},98 Q${cx},${r2(112 - 8 * e)} ${cx - 14},98 Z`;

  const br = g('br');
  const bi = g('bi');
  const bd = g('bd');
  const A: [number, number] = [124 - 7 * bi, 80 + 6 * bi + 8 * bd - 5 * br];
  const B: [number, number] = [140, 71 - 11 * br + 8 * bd + 3 * bi];
  const C: [number, number] = [156, 79 - 10 * br + 7 * bd - bi];

  const wf = g('flare') * 4;
  const wy = -3 * g('lu') + 3 * g('td');

  const sm = g('smile');
  const pu = g('purse');
  const cy = 170 - 7 * sm;
  const hw = 20 + 6 * g('out') - 7 * pu + 3 * Math.max(sm, 0);
  const Lx = 110 - hw;
  const Rx = 110 + hw;
  const ty = 166 - 4 * g('lu') - 3 * pu - Math.max(sm, 0);
  const by = 174 + 10 * g('ld') + 12 * g('jaw') - 3 * g('lup') + 3 * pu;

  const nlf = g('nlf');

  // حركات الرأس (tilt: ميلان جانبي، turn: دوران، fx: انحناء للأمام/تمدد للخلف)
  const tilt = g('tilt');
  const turn = g('turn');
  const fx = g('fx');
  const headT =
    `translate(${r2(110 + turn * 12)} ${r2(215 + fx * 7)}) rotate(${r2(tilt * 14)}) ` +
    `scale(${r2(1 - 0.05 * Math.abs(turn))} ${r2(1 - 0.03 * Math.abs(fx))}) translate(-110 -215)`;

  return {
    headT,

    hy: Math.max(-1, Math.min(1, g('hy'))),
    head,
    eyeL: eyePath(80),
    eyeR: eyePath(140),
    irisOpacity: clamp01(1 - e < 0.15 ? 0 : 1 - e),
    browR: `M${r2(A[0])},${r2(A[1])} Q${B[0]},${r2(B[1])} ${C[0]},${r2(C[1])}`,
    browL: `M${r2(220 - A[0])},${r2(A[1])} Q${220 - B[0]},${r2(B[1])} ${r2(220 - C[0])},${r2(C[1])}`,
    noseWingR: `M122,146 Q${r2(130 + wf)},${r2(152 + wy)} ${r2(126 + wf)},${r2(160 + wy)}`,

    noseBase: `M${r2(94 - wf)},${r2(160 + wy)} Q110,${r2(168 + wy)} ${r2(126 + wf)},${r2(160 + wy)}`,
    nostril: { lx: r2(102 - wf * 0.6), rx: r2(118 + wf * 0.6), y: r2(158.5 + wy), rad: r2(3 + g('flare') * 1.5) },

/** مفتاح العصب الأساسي للعضلة، أو null إن كانت مختلطة/غير واضحة */
export function nerveKeyOf(id: string): string | null {
  const m = MUSCLE_BY_ID[id];
  if (!m) return null;
  const n = m.nerve;
  const vii = /\bCN VII\b/.test(n);
  const v3 = /\bCN V3\b/.test(n);
  if (vii && v3) return null;
  if (vii) return 'vii';
  if (v3) return 'v3';
  if (/\bCN XI\b/.test(n)) return 'xi';
  if (/ansa cervicalis/i.test(n)) return 'ansa';
  if (/hypoglossal/i.test(n)) return 'c1xii';
  if (/C1/.test(n) && /anterior ramus of C1 \(/.test(n)) return 'c1xii';
  if (/dorsal scapular/i.test(n)) return 'dorsal';
  if (/posterior rami/i.test(n)) return 'post';
  if (/anterior rami/i.test(n)) return 'ant';
  return null;
}

const AMBIGUOUS_ACTION_GROUPS: readonly (readonly string[])[] = [
  ['lls', 'zmin', 'llsan', 'lao', 'zmaj'],
  ['masseter', 'temporalis'],
  ['corrugator', 'procerus'],
  ['scaleneAnt', 'scaleneMid', 'scalenePost'],
  ['digastric', 'stylohyoid', 'mylohyoid', 'geniohyoid'],
  ['omohyoid', 'sternohyoid', 'thyrohyoid', 'sternothyroid'],
  ['rcAnterior', 'rcLateralis', 'longusCapitis', 'longusColli'],
];

function shuffle<T>(arr: readonly T[]): T[] {
  const a = [...arr];
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    const t = a[i] as T;
    a[i] = a[j] as T;
    a[j] = t;
  }
  return a;
}

function sameAmbiguityGroup(a: string, b: string): boolean {
  return AMBIGUOUS_ACTION_GROUPS.some((g) => g.includes(a) && g.includes(b));
}

function inScope(id: string, scope: QuizScope): boolean {
  if (scope === 'all') return true;
  return scope === 'neck' ? NECK_MUSCLE_IDS.has(id) : !NECK_MUSCLE_IDS.has(id);
}

function distractors(id: string, n: number, kind: QuestionKind, scope: QuizScope): string[] {
  const target = MUSCLE_BY_ID[id];
  const base = MUSCLES.filter((m) => m.id !== id && !(kind === 'action' && sameAmbiguityGroup(id, m.id)));
  const scoped = base.filter((m) => inScope(m.id, scope));
  const pool = scoped.length >= n ? scoped : base;
  const same = shuffle(pool.filter((m) => m.region === target?.region));
  const other = shuffle(pool.filter((m) => m.region !== target?.region));
  return [...same, ...other].slice(0, n).map((m) => m.id);
}

function eligibleKinds(id: string): QuestionKind[] {
  const m = MUSCLE_BY_ID[id];
  const kinds: QuestionKind[] = ['locate', 'identify', 'action', 'attach'];
  if (m && branchesOf(m).length === 1) kinds.push('branch');
  if (nerveKeyOf(id)) kinds.push('nerve');
  return kinds;
}

function build(id: string, kind: QuestionKind, scope: QuizScope): Question {
  if (kind === 'locate') return { kind, id, options: [], correct: id };
  if (kind === 'branch') {
    const m = MUSCLE_BY_ID[id];
    const only = m ? branchesOf(m)[0] : undefined;
    return { kind, id, options: FACIAL_BRANCHES.map((b) => b.id), correct: only ?? 'buccal' };
  }
  if (kind === 'nerve') {
    const key = nerveKeyOf(id) ?? 'vii';
    const others = shuffle(NERVE_OPTIONS.filter((o) => o.id !== key)).slice(0, 3).map((o) => o.id);
    return { kind, id, options: shuffle([key, ...others]), correct: key };
  }
  return { kind, id, options: shuffle([id, ...distractors(id, 3, kind, scope)]), correct: id };
}

function triangleQuestion(): Question | null {
  const withBounds = TRIANGLES.filter((t) => (TRIANGLE_BOUNDARY[t.id]?.length ?? 0) > 0);
  const t = shuffle(withBounds)[0];
  if (!t) return null;
  if (Math.random() < 0.5) {
    const bounds = TRIANGLE_BOUNDARY[t.id] ?? [];
    const correct = shuffle(bounds)[0];
    if (!correct) return null;
    const wrong = shuffle(MUSCLES.filter((m) => !bounds.includes(m.id) && NECK_MUSCLE_IDS.has(m.id)).map((m) => m.id)).slice(0, 3);
    return { kind: 'boundary', id: correct, ref: t.id, options: shuffle([correct, ...wrong]), correct };
  }
  const correct = shuffle(t.contents)[0];
  if (!correct) return null;
  const foreign = TRIANGLES.filter((x) => x.id !== t.id).flatMap((x) => x.contents).filter((c) => !t.contents.includes(c));
  const wrong = shuffle([...new Set(foreign)]).slice(0, 3);
  if (wrong.length < 2) return null;
  return { kind: 'contents', id: t.id, ref: t.id, options: shuffle([correct, ...wrong]), correct };
}

export function makeRound(opts: RoundOptions | number = 10): Question[] {
  const o: RoundOptions = typeof opts === 'number' ? { count: opts, scope: 'all' } : opts;
  const pool = o.only && o.only.length > 0 ? [...o.only] : MUSCLES.map((m) => m.id).filter((id) => inScope(id, o.scope));
  const ids = shuffle(pool).slice(0, o.count);
  const out: Question[] = [];
  let prev: QuestionKind | null = null;
  const allowTri = (o.scope === 'neck' || o.scope === 'all') && !(o.only && o.only.length > 0);

  for (const id of ids) {
    if (allowTri && out.length > 0 && Math.random() < (o.scope === 'neck' ? 0.22 : 0.1)) {
      const tq = triangleQuestion();
      if (tq) {
        out.push(tq);
        prev = tq.kind;
        continue;
      }
    }
    const options = shuffle(eligibleKinds(id));
    const kind = options.find((k) => k !== prev) ?? options[0] ?? 'identify';
    out.push(build(id, kind, o.scope));
    prev = kind;
  }
  return out;
}

function synergistsOf(id: string): string[] {
  const m = MUSCLE_BY_ID[id];
  if (!m) return [];
  const b = branchesOf(m);
  if (b.length !== 1) return [];
  return MUSCLES.filter((x) => x.id !== id && branchesOf(x).includes(b[0] as never) && x.region === m.region).map((x) => x.id);
}

  const inn = m ? innervationOf(m) : null;
  if (!m || !inn) return [];
  return MUSCLES.filter((x) => x.id !== id && innervationOf(x) === inn && x.region !== m.region).map((x) => x.id);
}

    ...synergistsOf(id).slice(0, 2).map((x) => ({ id: x, kind: 'synergist' as const })),

  return out.filter((r) => !seen.has(r.id) && (seen.add(r.id), true)).slice(0, limit);

/**
 * مراجعة متباعدة مبسّطة (مستوحاة من SM-2).
 * نسيت / صعب (<3): تُعاد بعد 10 دقائق. بعدها: يوم، 3 أيام، 7 أيام، ثم الفاصل × معامل السهولة.
 */
export type Grade = 0 | 1 | 2 | 3 | 4 | 5;

export interface SrsCard {
  id: string;
  reps: number;
  ease: number;
  interval: number;
  due: number;
  last: number;
  total: number;
  lapses: number;
}

export interface SrsState {
  cards: Record<string, SrsCard>;
}

const DAY = 86_400_000;
const MIN = 60_000;

export function newCard(id: string, now = Date.now()): SrsCard {
  return { id, reps: 0, ease: 2.5, interval: 0, due: now, last: 0, total: 0, lapses: 0 };
}

export function grade(card: SrsCard, g: Grade, now = Date.now()): SrsCard {
  const c: SrsCard = { ...card, total: card.total + 1, last: now };
  if (g < 3) {
    c.reps = 0;
    c.interval = 0;
    c.due = now + 10 * MIN;
    c.ease = Math.max(1.3, card.ease - 0.2);
    c.lapses = card.lapses + 1;
    return c;
  }
  c.reps = card.reps + 1;
  if (c.reps === 1) {
    c.interval = 1;
  } else if (c.reps === 2) {
    c.interval = 3;
  } else if (c.reps === 3) {
    c.interval = 7;
  } else {
    const factor = g === 5 ? 1.3 : g === 4 ? 1.15 : 1;
    c.interval = Math.max(1, Math.round(card.interval * card.ease * factor));
  }
  c.due = now + c.interval * DAY;
  c.ease = Math.max(1.3, Math.min(3, card.ease + (0.1 - (5 - g) * 0.08)));
  return c;
}

export function isDue(card: SrsCard, now = Date.now()): boolean {
  return card.due <= now;
}

/**
 * سطح تقريبي للرأس والرقبة.
 * مصدر واحد لشكل الرقبة في الرسم 2D ولمجسّم الـ 3D:
 * يحوّل إحداثيات الرسم (x, y) إلى عمق z حتى تستند عضلات الـ 3D إلى نفس إحداثيات الـ SVG.

export const UNITS = 100;
/** y في الرسم يقابل Y = 0 في العالم */
export const Y0 = 330;

  [430, 84, 36, 54],
  [490, 96, 46, 58],
  [540, 106, 52, 62],
  [570, 126, 56, 66],
  [596, 166, 64, 74],
  [616, 196, 72, 82],
  [700, 198, 74, 84],
  [780, 198, 74, 84],

export function sampleTable(table: readonly SurfaceRow[], y: number): SurfaceRow {
  const first = table[0];
  const last = table[table.length - 1];
  if (!first || !last) return [y, 0, 0, 0];
  if (y <= first[0]) return [y, first[1], first[2], first[3]];
  if (y >= last[0]) return [y, last[1], last[2], last[3]];
  for (let i = 1; i < table.length; i++) {
    const b = table[i];
    const a = table[i - 1];
    if (!a || !b || y > b[0]) continue;
    const t = (y - a[0]) / (b[0] - a[0] || 1);
    return [y, a[1] + (b[1] - a[1]) * t, a[2] + (b[2] - a[2]) * t, a[3] + (b[3] - a[3]) * t];
  }
  return [y, last[1], last[2], last[3]];
}

const smooth = (t: number): number => t * t * (3 - 2 * t);
const clamp01 = (t: number): number => Math.max(0, Math.min(1, t));

function ellipseZ(w: number, depth: number, x: number): number {
  if (w <= 0.01 || depth <= 0.01) return 0;
  const u = Math.max(-0.97, Math.min(0.97, (x - CX) / w));
  return depth * Math.sqrt(1 - u * u);
}

/** منطقة المزج بين الرأس والرقبة (تحت الذقن) */
const BLEND_FROM = 412;
const BLEND_TO = 440;

/** عمق السطح الأمامي عند النقطة (x, y) بوحدات الرسم */
export function surfaceZ(x: number, y: number): number {
  const h = sampleTable(HEAD_TABLE, y);
  const n = sampleTable(NECK_TABLE, y);
  const zh = ellipseZ(h[1], h[2], x);
  const zn = ellipseZ(n[1], n[2], x);
  if (y <= BLEND_FROM) return zh;
  if (y >= BLEND_TO) return zn;
  const t = smooth(clamp01((y - BLEND_FROM) / (BLEND_TO - BLEND_FROM)));
  return zh * (1 - t) + zn * t;
}

/** إحداثيات العالم (three.js) لنقطة من الرسم رُفعت `lift` وحدة عن السطح */
export function toWorld(x: number, y: number, z: number): [number, number, number] {
  return [(x - CX) / UNITS, (Y0 - y) / UNITS, z / UNITS];
}

export function surfacePoint(x: number, y: number, lift: number): [number, number, number] {
  return toWorld(x, y, surfaceZ(x, y) + lift);
}

function neckSide(sign: 1 | -1, ys: readonly number[]): [number, number][] {
  return ys.map((y) => [CX + sign * sampleTable(NECK_TABLE, y)[1], y]);
}

function buildNeckOutline(): string {
  const ys: number[] = [];
  for (let y = 380; y <= 700; y += 16) ys.push(y);
  const right = neckSide(1, ys);
  const left = neckSide(-1, ys).reverse();
  const pts = [...right, ...left];
  return (
    pts.map(([x, y], i) => `${i === 0 ? 'M' : 'L'}${Math.round(x * 10) / 10},${y}`).join('') + 'Z'
  );
}

/** مسار الرقبة والكتفين (يُرسم خلف الرأس) */
export const NECK_OUTLINE: string = buildNeckOutline();

  ay: number;

interface Raw {
  m: Muscle;
  ax: number;
  ay: number;
  ly: number;
}

export function layoutLabels(
  vb: ViewBox,
  widthPx: number,
  emWidth: (text: string) => number,
): LabelLayout {
  if (!(widthPx > 0)) return { fs: 0, items: [] };

  const s = widthPx / vb[2];
  const fs = 11 / s;
  const lineH = fs * 1.45;
  const lists: [Raw[], Raw[]] = [[], []];

  for (const m of MUSCLES) {
    const ax = m.label.s === 'L' ? W0 - m.label.x : m.label.x;
    const ay = m.label.y;
    if (ax < vb[0] + fs || ax > vb[0] + vb[2] - fs || ay < vb[1] + fs || ay > vb[1] + vb[3] - fs) continue;
    lists[m.label.s === 'L' ? 0 : 1].push({ m, ax, ay, ly: ay });
  }

  const items: PlacedLabel[] = [];

  lists.forEach((list, si) => {
    list.sort((a, b) => a.ay - b.ay);

    for (let i = 1; i < list.length; i++) {
      const cur = list[i];
      const prev = list[i - 1];
      if (cur && prev && cur.ly < prev.ly + lineH) cur.ly = prev.ly + lineH;
    }

    const maxY = vb[1] + vb[3] - fs;
    for (let i = list.length - 1; i >= 0; i--) {
      const cur = list[i];
      if (!cur) continue;
      const lim = maxY - (list.length - 1 - i) * lineH;
      if (cur.ly > lim) cur.ly = lim;
    }

    for (const it of list) {
      const text = it.m.short ?? it.m.en;
      const x = si === 0 ? vb[0] + fs * 0.5 : vb[0] + vb[2] - fs * 0.5;
      const len = emWidth(text) * fs;
      items.push({
        id: it.m.id,
        text,
        ax: it.ax,
        ay: it.ay,
        tx: x,
        ty: it.ly + fs * 0.35,
        anchor: si === 0 ? 'start' : 'end',
        sx: si === 0 ? x + len + fs * 0.4 : x - len - fs * 0.4,
        sy: it.ly,
      });
    }
  });

  return { fs, items };
}

  url: string,

export interface AccessToken {
  slug: string;
  label: string | null;
  is_active: boolean;
  study_year: string;
  expires_at: string | null;
}

/**
 * يجيب رابط الوصول المؤقت (slug) إذا كان فعّالاً
 */
export async function getAccessToken(
  slug: string
): Promise<AccessToken | null> {
  const { data } = await supabaseAdmin
    .from('sg_access_tokens')
    .select('*')
    .eq('slug', slug)
    .maybeSingle();

  if (!data) return null;
  if (!data.is_active) return null;
  if (data.expires_at && new Date(data.expires_at).getTime() < Date.now())
    return null;

  return data as AccessToken;
}

/**
 * يجيب إعدادات النظام (صف واحد فقط)
 */
export async function getSettings(): Promise<Settings> {
  const { data } = await supabaseAdmin
    .from('sg_settings')
    .select('*')
    .eq('id', 1)
    .maybeSingle();

  if (!data) {
    return {
      id: 1,
      target_size: 10,
      preferred_male: 5,
      preferred_female: 5,
      registration_open: true,
      allow_create: true,
      allow_join: true,
      session_expires_at: null,
      updated_at: new Date().toISOString(),
    };
  }
  return data as Settings;
}

export function isRegistrationOpen(s: Settings): boolean {
  return s.registration_open === true;
}

const TOKEN_KEY = 'sg_token';

export function saveToken(token: string): void {
  if (typeof window === 'undefined') return;
  localStorage.setItem(TOKEN_KEY, token);
}

export function getToken(): string | null {
  if (typeof window === 'undefined') return null;
  return localStorage.getItem(TOKEN_KEY);
}

export function clearToken(): void {
  if (typeof window === 'undefined') return;
  localStorage.removeItem(TOKEN_KEY);
}

interface ApiBody {
  ok?: boolean;
  error?: string;
  data?: unknown;
}

export async function groupsFetch<T = unknown>(
  endpoint: string,
  body: Record<string, unknown> = {}
): Promise<T> {
  const token = getToken();
  const res = await fetch(endpoint, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
    },
    body: JSON.stringify(body),
  });

  const json: ApiBody = await res.json().catch(() => ({}));

  if (!res.ok) {
    throw new Error(json?.error || `فشل الاتصال (${res.status})`);
  }
  if (json?.ok === false) {
    throw new Error(json?.error || 'حدث خطأ');
  }
  return (json?.data ?? json) as T;
}

import { randomBytes } from 'node:crypto';
import { supabaseAdmin } from './supabase-admin';
import type { Student } from '@/lib/types/groups';

// بغداد = UTC+3 (بدون توقيت صيفي)
const BAGHDAD_OFFSET_MS = 3 * 60 * 60 * 1000;

/**
 * يرجع "الآن" بتوقيت بغداد (كـ Date نتعامل معه عبر getUTC*)
 */
function getBaghdadNow(): Date {
  return new Date(Date.now() + BAGHDAD_OFFSET_MS);
}

/**
 * يحسب موعد "الأحد القادم 23:59:59 بتوقيت بغداد"
 * - إذا اليوم أحد وقبل 23:59 → نفس اليوم
 * - إذا اليوم أحد وبعد 23:59 → الأحد القادم
 * - أي يوم آخر → الأحد القادم
 */
export function getNextSundayDeadline(): Date {
  const baghdadNow = getBaghdadNow();
  const day = baghdadNow.getUTCDay(); // 0 = الأحد
  const daysUntilSunday = (7 - day) % 7;

  const target = new Date(baghdadNow);
  target.setUTCDate(target.getUTCDate() + daysUntilSunday);
  target.setUTCHours(23, 59, 59, 999);

  if (target.getTime() <= baghdadNow.getTime()) {
    target.setUTCDate(target.getUTCDate() + 7);
  }

  // نرجع للـ UTC الحقيقي
  return new Date(target.getTime() - BAGHDAD_OFFSET_MS);
}

/**
 * تاريخ انتهاء الجلسة:
 * - إذا الأدمن حدد موعداً في sg_settings → نستخدمه
 * - وإلا → الأحد القادم 23:59
 */
export async function getSessionExpiresAt(): Promise<Date> {
  const { data: settings } = await supabaseAdmin
    .from('sg_settings')
    .select('session_expires_at')
    .eq('id', 1)
    .maybeSingle();

  if (settings?.session_expires_at) {
    const d = new Date(settings.session_expires_at);
    if (!isNaN(d.getTime())) return d;
  }
  return getNextSundayDeadline();
}

export function generateToken(): string {
  return randomBytes(24).toString('hex');
}

 * ينشئ جلسة جديدة ويحفظها في DB، ويرجع التوكن
 */

  const token = generateToken();
  const expiresAt = await getSessionExpiresAt();

  const { error } = await supabaseAdmin.from('sg_sessions').insert({
    token,
    student_id: studentId,
    expires_at: expiresAt.toISOString(),

```tsx
function getFingerprint(): string {
  try {
    let fp = localStorage.getItem(FP_KEY);
    if (!fp) {
      fp = `${Date.now()}-${Math.random().toString(36).slice(2)}`;
      localStorage.setItem(FP_KEY, fp);
    }
    return fp;
  } catch {
    return '';
  }
}

  onReported?: () => void;

      setNoteText('');
      setDone(false);
```

```tsx
// components/Reveal.tsx
'use client';

import {
  useEffect,
  useRef,
  type CSSProperties,
  type ElementType,
  type ReactNode,
} from 'react';

type RevealTag = 'div' | 'li' | 'section' | 'article';

interface RevealProps {
  children: ReactNode;
  as?: RevealTag;
  /** تأخير الظهور بالمللي ثانية (للـ stagger) */
  delay?: number;
  /** مقدار الإزاحة الرأسية للأسفل قبل الظهور */
  y?: number;
  className?: string;
  style?: CSSProperties;
}

export function Reveal({
  children,
  as: Tag = 'div',
  delay = 0,
  y = 12,
  className = '',
  style,
}: RevealProps) {
  const ref = useRef<HTMLElement | null>(null);
  const Component: ElementType = Tag;

  useEffect(() => {
    const el = ref.current;
    if (!el) return;

    // احترم تفضيل تقليل الحركة — أظهر مباشرة
    if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
      el.dataset.revealed = 'true';
      return;
    }

    const observer = new IntersectionObserver(
      ([entry]) => {
        if (entry.isIntersecting) {
          el.dataset.revealed = 'true';
          observer.disconnect();
        }
      },
      {
        threshold: 0.08,
        rootMargin: '0px 0px -40px 0px',
      },
    );

    observer.observe(el);
    return () => observer.disconnect();
  }, []);

  const combinedStyle = {
    ...style,
    '--reveal-delay': `${delay}ms`,
    '--reveal-y': `${y}px`,
  } as CSSProperties;

  // ✅ الحل: نستخدم callback ref بدل ref object
  // لأن Component قد يكون div / li / section / article
  // وTypeScript لا يستطيع توحيد أنواع الـrefs لهذه الحالات.
  // الـcallback ref يعمل مع جميع الاحتمالات (contravariance).
  const setRef = (el: HTMLElement | null) => {
    ref.current = el;
  };

  return (
    <Component
      ref={setRef}
      className={`reveal-scroll ${className}`.trim()}
      style={combinedStyle}
    >
      {children}
    </Component>
  );
}
```

```tsx
// components/ThemeProvider.tsx
'use client';

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useState,
  type ReactNode,
} from 'react';

type Theme = 'light' | 'dark';

interface ThemeContextValue {
  theme: Theme;
  toggleTheme: () => void;
  setTheme: (theme: Theme) => void;
}

const ThemeContext = createContext<ThemeContextValue | undefined>(undefined);

export function ThemeProvider({ children }: { children: ReactNode }) {
  const [theme, setThemeState] = useState<Theme>('light');

  // ===== تحميل الثيم المحفوظ أو المفضّل =====
  // ===== تحميل الثيم المحفوظ — الوضع النهاري افتراضياً =====
useEffect(() => {
  try {
    const saved = localStorage.getItem('theme') as Theme | null;
    const initial: Theme = saved ?? 'light';
    setThemeState(initial);
    document.documentElement.classList.toggle('dark', initial === 'dark');
  } catch {
    /* في حال فشل localStorage */
  }
}, []);

  // ===== تغيير الثيم =====
  const setTheme = useCallback((next: Theme) => {
    setThemeState(next);
    try {
      localStorage.setItem('theme', next);
    } catch {
      /* تجاهل */
    }
    document.documentElement.classList.toggle('dark', next === 'dark');
  }, []);

  const toggleTheme = useCallback(() => {
    setTheme(theme === 'light' ? 'dark' : 'light');
  }, [theme, setTheme]);

  return (
    <ThemeContext.Provider value={{ theme, toggleTheme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme(): ThemeContextValue {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme must be used within ThemeProvider');
  return ctx;
}
```

```tsx
const CYCLE = 3400;
const ease = (x: number): number => (x < 0.5 ? 2 * x * x : 1 - Math.pow(-2 * x + 2, 2) / 2);

function kAt(t: number): number {
  const c = t % CYCLE;
  if (c < 800) return ease(c / 800);
  if (c < 1900) return 1;
  if (c < 2600) return 1 - ease((c - 1900) / 700);
  return 0;
}

function phaseForK(k: number): number {
  const x = k < 0.5 ? Math.sqrt(k / 2) : 1 - Math.sqrt((1 - k) / 2);
  return x * 800;
}
```

```tsx
function Arrow({ a, mirror }: { a: readonly [number, number, number, number]; mirror: boolean }) {
  const [x1, y1, x2, y2] = a;
  const ang = Math.atan2(y2 - y1, x2 - x1);
  const s = 8;
  const ex = x2 - 2.5 * Math.cos(ang);
  const ey = y2 - 2.5 * Math.sin(ang);
  const h1x = x2 - s * Math.cos(ang - 0.45);
  const h1y = y2 - s * Math.sin(ang - 0.45);
  const h2x = x2 - s * Math.cos(ang + 0.45);
  const h2y = y2 - s * Math.sin(ang + 0.45);
  const line = `M${x1},${y1} L${ex.toFixed(1)},${ey.toFixed(1)}`;
  return (
    <g transform={mirror ? MIRROR : undefined}>
      <path d={line} className="fa-halo" />
      <path d={line} className="fa-arrow" />
      <path d={`M${x2},${y2} L${h1x.toFixed(1)},${h1y.toFixed(1)} L${h2x.toFixed(1)},${h2y.toFixed(1)} Z`} className="fa-ahead" />
    </g>
  );
}

function ArrowsLayer({ ids }: { ids: readonly string[] }) {
  return (
    <>
      {ids.flatMap((id) => {
        const m = MUSCLE_BY_ID[id];
        if (!m) return [];
        return m.arrows.flatMap((a, i) => {
          const out = [<Arrow key={`${id}-${i}-0`} a={a} mirror={false} />];
          if (m.paired) out.push(<Arrow key={`${id}-${i}-1`} a={a} mirror />);
          return out;
        });
      })}
    </>
  );
}

  // ----- مراجع -----
  const svgRef = useRef<SVGSVGElement>(null);
  const figRef = useRef<HTMLDivElement>(null);
  const tipRef = useRef<HTMLDivElement>(null);
  const cardRef = useRef<HTMLElement>(null);
  const measureRef = useRef<SVGTextElement>(null);
  const vbRef = useRef<ViewBox>(vb);
  const labelsOnRef = useRef(false);
  const animRef = useRef(0);
  const reducedRef = useRef(false);
  const interacted = useRef(false);
  const bestRef = useRef(0);

  // ==================== الكاميرا ====================
  const animateTo = useCallback((to: ViewBox, instant = false) => {
    cancelAnimationFrame(animRef.current);
    const from = vbRef.current;
    if (instant || reducedRef.current) {
      vbRef.current = to;
      setVb(to);
      return;
    }
    const t0 = performance.now();
    const step = (t: number) => {
      const k = Math.min(1, (t - t0) / 450);
      const e = 1 - Math.pow(1 - k, 3);
      const mix = (i: 0 | 1 | 2 | 3): number => from[i] + (to[i] - from[i]) * e;
      const next: ViewBox = [mix(0), mix(1), mix(2), mix(3)];
      vbRef.current = next;
      setVb(next);
      if (k < 1) animRef.current = requestAnimationFrame(step);
    };
    animRef.current = requestAnimationFrame(step);
  }, []);

  const changeZoom = useCallback(
    (z: ZoomId, instant = false) => {
      setZoom(z);
      animateTo(viewBoxFor(z, labelsOnRef.current), instant);
    },
    [animateTo],
  );

  const toggleLabels = useCallback(() => {
    const next = !labelsOnRef.current;
    labelsOnRef.current = next;
    setLabelsOn(next);
    animateTo(viewBoxFor(zoom, next), true);
  }, [animateTo, zoom]);

  useEffect(() => () => cancelAnimationFrame(animRef.current), []);

  // ==================== بدء التشغيل ====================
  useIsoLayoutEffect(() => {
    try {
      reducedRef.current = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    } catch {
      /* تجاهل */
    }

    let wantLabels = false;
    try {
      // إخفاء التسميات افتراضياً على الجوال (أقل من 1024px)
      wantLabels = window.matchMedia('(min-width: 1024px)').matches;
    } catch {
      /* تجاهل */
    }

    let startZoom: ZoomId = 'whole';
    try {
      const p = new URLSearchParams(window.location.search);
      const mid = p.get('m');
      const xid = p.get('x');
      if (xid && EXPRESSIONS.some((e) => e.id === xid)) {
        setExpressionId(xid);
        setSelectedId(null);
      } else if (mid && MUSCLE_BY_ID[mid]) {
        setSelectedId(mid);
        startZoom = MUSCLE_BY_ID[mid].zoom;
      }
    } catch {
      /* تجاهل */
    }

    labelsOnRef.current = wantLabels;
    setLabelsOn(wantLabels);
    setZoom(startZoom);
    const v = viewBoxFor(startZoom, wantLabels);
    vbRef.current = v;
    setVb(v);
  }, []);

  useEffect(() => {
    try {
      const n = Number(localStorage.getItem(BEST_KEY));
      if (Number.isFinite(n) && n > 0) {
        bestRef.current = n;
        setBest(n);
      }
    } catch {
      /* تجاهل */
    }
  }, []);

  useEffect(() => {
    if (!interacted.current || mode !== 'explore') return;
    try {
      const p = new URLSearchParams();
      if (expressionId) p.set('x', expressionId);
      else if (selectedId) p.set('m', selectedId);
      const qs = p.toString();
      window.history.replaceState(null, '', qs ? `?${qs}` : window.location.pathname);
    } catch {
      /* تجاهل */
    }
  }, [selectedId, expressionId, mode]);

  // ==================== القياسات ====================
  useEffect(() => {
    const el = svgRef.current;
    if (!el) return;
    setSvgW(el.getBoundingClientRect().width);
    if (typeof ResizeObserver === 'undefined') return;
    const ro = new ResizeObserver((entries) => {
      const w = entries[0]?.contentRect.width;
      if (w) setSvgW(w);
    });
    ro.observe(el);
    return () => ro.disconnect();
  }, []);

  useEffect(() => {
    let cancelled = false;
    const run = () => {
      const t = measureRef.current;
      if (!t || cancelled || typeof t.getComputedTextLength !== 'function') return;
      const next: Record<string, number> = {};
      for (const m of MUSCLES) {
        const text = m.short ?? m.en;
        t.textContent = text;
        const len = t.getComputedTextLength();
        if (len > 0) next[text] = len / 100;
      }
      t.textContent = '';
      if (Object.keys(next).length) setEmW(next);
    };
    run();
    try {
      void document.fonts?.ready.then(run);
    } catch {
      /* تجاهل */
    }
    return () => {
      cancelled = true;
    };
  }, []);

  // ==================== التحديد ====================
  const scrollToFigure = useCallback(() => {
    const fig = figRef.current;
    const card = cardRef.current;
    if (!fig) return;
    const navH = parseFloat(getComputedStyle(document.documentElement).getPropertyValue('--nav-h')) || 0;
    let stuck = 0;
    if (card && getComputedStyle(card).position === 'sticky' && window.matchMedia('(max-width: 879px)').matches) {
      stuck = card.offsetHeight;
    }
    const top = window.scrollY + fig.getBoundingClientRect().top - navH - stuck - 12;
    window.scrollTo({ top: Math.max(0, top), behavior: reducedRef.current ? 'auto' : 'smooth' });

      const q = prev.questions[prev.index];
      if (!q) return prev;

        picked: choice,

        bestStreak: Math.max(prev.bestStreak, streak),

  const expression = expressionId ? (EXPRESSIONS.find((e) => e.id === expressionId) ?? null) : null;

    if (mode === 'quiz') return null;
    if (expression) return new Set(expression.muscles.map((x) => x.id));
    if (branch === 'v3') return new Set(V3_IDS);

          ) : (
            <>

                <div className="min-w-0">
                  <p className="text-[11.5px] font-bold tracking-wide text-ink-muted">
                    {expression ? 'Expression' : selected?.region}
                  </p>

                      <h2 dir="ltr" className="mt-0.5 text-start font-display text-2xl font-bold leading-tight text-ink">

  );
}
```

## components\anatomy\QuizPanel.tsx

```tsx
'use client';

// components/anatomy/QuizPanel.tsx
import { MUSCLE_BY_ID } from '@/lib/anatomy-data';
import { ACTION_AR } from '@/lib/anatomy-ar';
import { FACIAL_BRANCHES } from '@/lib/anatomy-extras';
import type { Question } from '@/lib/anatomy-quiz';
import { badge, btnGhost, btnPrimary } from './ui-classes';

export interface QuizState {
  questions: Question[];
  index: number;
  picked: string | null;
  score: number;
  streak: number;
  bestStreak: number;
  missed: string[];
  done: boolean;
}

interface Props {
  state: QuizState;
  best: number;
  onAnswer: (choice: string) => void;
  onNext: () => void;
  onRestart: () => void;
  onExit: () => void;
  onReview: (id: string) => void;
}

const PROMPT: Record<Question['kind'], string> = {
  locate: 'اضغط على هذه العضلة في الرسم',
  identify: 'ما اسم العضلة المظلَّلة في الرسم؟',
  action: 'أي عضلة تقوم بهذه الوظيفة؟',
  attach: 'أي عضلة لها هذا المنشأ والمرتكز؟',
  branch: 'ما فرع العصب الوجهي الذي يغذّي هذه العضلة؟',
};

const optionBase =
  'w-full rounded-field border px-3 py-2 text-start text-sm transition-colors focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-teal disabled:cursor-default motion-reduce:transition-none';

function optionTone(answered: boolean, isCorrect: boolean, isPicked: boolean): string {
  if (!answered) return 'border-line bg-paper-soft text-ink hover:border-teal/50';
  if (isCorrect) return 'border-teal bg-teal-soft text-ink';
  if (isPicked) return 'border-coral bg-coral-soft text-ink';
  return 'border-line bg-paper-soft text-ink-muted opacity-60';
}

export default function QuizPanel({ state, best, onAnswer, onNext, onRestart, onExit, onReview }: Props) {
  const total = state.questions.length;

  if (state.done) {
    const pct = Math.round((state.score / total) * 100);
    const verdict =
      pct === 100 ? 'ممتاز! علامة كاملة.' : pct >= 80 ? 'مستواك قوي، راجع الأخطاء وتنتهي.' : pct >= 50 ? 'جيد، يحتاج مراجعة بسيطة.' : 'لا بأس، ابدأ بالاستكشاف ثم أعد المحاولة.';
    return (
      <div>
        <p className="text-xs font-bold tracking-wide text-ink-muted">انتهت الجولة</p>
        <p className="mt-1 font-display text-3xl font-bold text-ink">
          {state.score}
          <span className="text-lg text-ink-muted"> / {total}</span>
        </p>
        <p className="mt-1 text-sm text-ink-soft">{verdict}</p>
        <div className="mt-2 flex flex-wrap gap-1.5">
          <span className={badge}>أفضل نتيجة: {Math.max(best, state.score)}/{total}</span>
          <span className={badge}>أطول سلسلة: {state.bestStreak}</span>
        </div>

        {state.missed.length > 0 && (
          <div className="mt-3">
            <p className="text-xs font-bold text-ink-muted">للمراجعة (اضغط لفتحها في الأطلس):</p>
            <div className="mt-1.5 flex flex-wrap gap-1.5">
              {state.missed.map((id) => (
                <button
                  key={id}
                  type="button"
                  dir="ltr"
                  onClick={() => onReview(id)}
                  className="rounded-chip border border-coral/40 bg-coral-soft px-2.5 py-1 text-xs text-ink hover:border-coral focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-teal"
                >
                  {MUSCLE_BY_ID[id]?.en ?? id}
                </button>
              ))}
            </div>
          </div>
        )}

        <div className="mt-4 flex flex-wrap gap-2">
          <button type="button" className={btnPrimary} onClick={onRestart}>جولة جديدة</button>
          <button type="button" className={btnGhost} onClick={onExit}>رجوع للأطلس</button>
        </div>
      </div>
    );
  }

  const q = state.questions[state.index];
  if (!q) return null;
  const m = MUSCLE_BY_ID[q.id];
  const answered = state.picked !== null;
  const correct = answered && state.picked === q.correct;

  return (
    <div>
      <div className="flex items-center justify-between gap-2">
        <p className="text-xs font-bold tracking-wide text-ink-muted">
          سؤال <span className="font-mono">{state.index + 1}</span> من <span className="font-mono">{total}</span>
        </p>
        <div className="flex items-center gap-1.5">
          <span className={badge}>النتيجة {state.score}</span>
          {state.streak >= 2 && <span className={badge}>🔥 {state.streak}</span>}
          <button type="button" onClick={onExit} className="text-xs font-semibold text-ink-muted underline-offset-2 hover:text-teal hover:underline focus-visible:outline-2 focus-visible:outline-teal">
            خروج
          </button>
        </div>
      </div>

      <div className="mt-2 h-1 overflow-hidden rounded-full bg-line" aria-hidden="true">
        <div className="h-full bg-teal transition-all duration-300 motion-reduce:transition-none" style={{ width: `${((state.index + (answered ? 1 : 0)) / total) * 100}%` }} />
      </div>

      <h3 className="mt-3 text-sm font-bold text-ink">{PROMPT[q.kind]}</h3>

      {(q.kind === 'locate' || q.kind === 'branch') && m && (
        <p dir="ltr" className="mt-1 text-start font-display text-xl font-bold text-ink">{m.en}</p>
      )}
      {q.kind === 'action' && m && (
        <p dir="ltr" className="mt-1 rounded-field border border-line bg-paper px-3 py-2 text-start text-[13px] leading-snug text-ink-soft">{m.actEn}</p>
      )}
      {q.kind === 'attach' && m && (
        <dl dir="ltr" className="mt-1 rounded-field border border-line bg-paper px-3 py-2 text-start text-[13px] leading-snug text-ink-soft">
          <dt className="text-[11px] font-bold uppercase tracking-wide text-ink-muted">Origin</dt>
          <dd>{m.origin}</dd>
          <dt className="mt-1 text-[11px] font-bold uppercase tracking-wide text-ink-muted">Insertion</dt>
          <dd>{m.insertion}</dd>
        </dl>
      )}

      {q.kind === 'locate' && !answered && (
        <button type="button" className={`${btnGhost} mt-3 w-full`} onClick={() => onAnswer('__skip')}>
          لا أعرف
        </button>
      )}

      {q.kind !== 'locate' && (
        <ul className="mt-3 grid gap-2">
          {q.options.map((opt) => {
            const label =
              q.kind === 'branch'
                ? FACIAL_BRANCHES.find((b) => b.id === opt)
                : null;
            const text = label ? label.en : (MUSCLE_BY_ID[opt]?.en ?? opt);
            return (
              <li key={opt}>
                <button
                  type="button"
                  dir="ltr"
                  disabled={answered}
                  onClick={() => onAnswer(opt)}
                  className={`${optionBase} ${optionTone(answered, opt === q.correct, opt === state.picked)}`}
                >
                  {text}
                  {label && <span dir="rtl" className="ms-2 text-xs text-ink-muted">({label.ar})</span>}
                </button>
              </li>
            );
          })}
        </ul>
      )}

      {answered && m && (
        <div className="mt-3 rounded-field border border-line bg-paper p-3" role="status">
          <p className={`text-sm font-bold ${correct ? 'text-teal' : 'text-coral-ink'}`}>
            {correct ? 'إجابة صحيحة ✓' : 'ليست صحيحة'}
          </p>
          <p dir="ltr" className="mt-1 text-start text-sm font-bold text-ink">{m.en}</p>
          <p dir="ltr" className="mt-0.5 text-start text-[13px] leading-relaxed text-ink">{m.actEn}</p>
          <p className="mt-0.5 text-[12px] leading-relaxed text-ink-muted">{ACTION_AR[m.id]}</p>
          <p dir="ltr" className="mt-1 text-start text-xs text-ink-muted">{m.nerve}</p>
        </div>
      )}

      {answered && (
        <button type="button" className={`${btnPrimary} mt-3 w-full`} onClick={onNext} autoFocus>
          {state.index + 1 >= total ? 'عرض النتيجة' : 'السؤال التالي'}
        </button>
      )}
    </div>
  );
}
```

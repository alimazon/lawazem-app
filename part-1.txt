import { memo, useEffect, useId, useMemo, useRef, useState } from 'react';
import { computeFace, scalePose, type Pose } from '@/lib/anatomy-pose';

const CYCLE = 3400;
const ease = (x: number): number => (x < 0.5 ? 2 * x * x : 1 - Math.pow(-2 * x + 2, 2) / 2);

function kAt(t: number): number {
  const c = t % CYCLE;
  if (c < 800) return ease(c / 800);
  if (c < 1900) return 1;
  if (c < 2600) return 1 - ease((c - 1900) / 700);
  return 0;
}

function phaseForK(k: number): number {
  const x = k < 0.5 ? Math.sqrt(k / 2) : 1 - Math.sqrt((1 - k) / 2);
  return x * 800;
}

interface Props {
  pose: Pose;
  poseKey: string;
  onK?: (k: number) => void;
}

function ExpressionPreview({ pose, poseKey, onK }: Props) {
  const uid = useId().replace(/:/g, '');
  const clipL = `${uid}-cl`;
  const clipR = `${uid}-cr`;

  const [playing, setPlaying] = useState(true);
  const [k, setK] = useState(0);
  const kRef = useRef(0);
  const lastKey = useRef(poseKey);

  useEffect(() => {
    try {
      if (window.matchMedia('(prefers-reduced-motion: reduce)').matches) {
        setPlaying(false);
        kRef.current = 1;
        setK(1);
      }
    } catch {
      /* تجاهل */
    }
  }, []);

  useEffect(() => {
    if (!playing) return;
    const resume = lastKey.current === poseKey;
    lastKey.current = poseKey;
    const start = performance.now() - (resume ? phaseForK(kRef.current) : 0);
    let raf = 0;
    const frame = (now: number) => {
      const next = kAt(now - start);
      if (Math.abs(next - kRef.current) > 0.002) {
        kRef.current = next;
        setK(next);
      }
      raf = requestAnimationFrame(frame);
    };
    raf = requestAnimationFrame(frame);
    return () => cancelAnimationFrame(raf);
  }, [playing, poseKey]);

  useEffect(() => {
    onK?.(k);
  }, [k, onK]);

  const f = useMemo(() => computeFace(scalePose(pose, k)), [pose, k]);
  const contracted = k > 0.5;

  return (
    <div className="fa-plate fa-preview">
      <svg viewBox="0 0 220 270" aria-hidden="true" style={{ direction: 'ltr' }}>
        <defs>
          <clipPath id={clipL}><path d={f.eyeL} /></clipPath>
          <clipPath id={clipR}><path d={f.eyeR} /></clipPath>
        </defs>

        <path d="M84,206 L82,270" className="fa-pl" />
        <path d="M136,206 L138,270" className="fa-pl" />
        <ellipse cx="36" cy="112" rx="5" ry="14" className="fa-pl" style={{ fill: 'var(--fa-skin)' }} />
        <ellipse cx="184" cy="112" rx="5" ry="14" className="fa-pl" style={{ fill: 'var(--fa-skin)' }} />
        <path d={f.head} className="fa-pl" style={{ fill: 'var(--fa-skin)' }} />

        <g style={{ opacity: f.op.neck }}>
          <path className="fa-pw" d="M92,214 L86,262" />
          <path className="fa-pw" d="M100,218 L96,262" />
          <path className="fa-pw" d="M128,214 L134,262" />
          <path className="fa-pw" d="M120,218 L124,262" />
        </g>
        <g style={{ opacity: f.op.forehead }}>
          <path className="fa-pw" d="M70,44 Q110,40 150,44" />
          <path className="fa-pw" d="M66,54 Q110,50 154,54" />
          <path className="fa-pw" d="M66,64 Q110,60 154,64" />
        </g>
        <g style={{ opacity: f.op.glabella }}>
          <path className="fa-pw" d="M104,78 L104.5,92" />
          <path className="fa-pw" d="M116,78 L115.5,92" />

function Arrow({ a, mirror }: { a: readonly [number, number, number, number]; mirror: boolean }) {
  const [x1, y1, x2, y2] = a;
  const ang = Math.atan2(y2 - y1, x2 - x1);
  const s = 8;
  const ex = x2 - 2.5 * Math.cos(ang);
  const ey = y2 - 2.5 * Math.sin(ang);
  const h1x = x2 - s * Math.cos(ang - 0.45);
  const h1y = y2 - s * Math.sin(ang - 0.45);
  const h2x = x2 - s * Math.cos(ang + 0.45);
  const h2y = y2 - s * Math.sin(ang + 0.45);
  const line = `M${x1},${y1} L${ex.toFixed(1)},${ey.toFixed(1)}`;
  return (
    <g transform={mirror ? MIRROR : undefined}>
      <path d={line} className="fa-halo" />
      <path d={line} className="fa-arrow" />
      <path d={`M${x2},${y2} L${h1x.toFixed(1)},${h1y.toFixed(1)} L${h2x.toFixed(1)},${h2y.toFixed(1)} Z`} className="fa-ahead" />
    </g>
  );
}

export type Innervation = 'VII' | 'V3';

export function innervationOf(m: Muscle): Innervation | null {
  if (m.nerve.includes('CN VII')) return 'VII';
  if (m.nerve.includes('CN V3')) return 'V3';
  return null;
}

export function branchesOf(m: Muscle): FacialBranch[] {
  if (innervationOf(m) !== 'VII') return [];
  const n = m.nerve.toLowerCase();
  return FACIAL_BRANCHES.filter((b) => n.includes(b.match)).map((b) => b.id);
}

export function pharyngealArchOf(m: Muscle): 1 | 2 | null {
  const inn = innervationOf(m);
  return inn === 'VII' ? 2 : inn === 'V3' ? 1 : null;
}

export const MUSCLES_BY_BRANCH: Record<FacialBranch, string[]> = (() => {
  const out: Record<FacialBranch, string[]> = {
    temporal: [], zygomatic: [], buccal: [], marginal: [], cervical: [],
  };
  for (const m of MUSCLES) for (const b of branchesOf(m)) out[b].push(m.id);
  return out;
})();

  if (error) throw new Error('فشل إنشاء الجلسة: ' + error.message);
  return token;
}

/**
 * يقرأ التوكن من Authorization header ويجيب الطالب صاحب الجلسة.
 * يرجع null لو ما فيه توكن، أو منتهي، أو غير موجود.
 */
export async function getStudentFromRequest(
  req: Request
): Promise<Student | null> {
  const auth = req.headers.get('authorization') || '';
  const m = auth.match(/^Bearer\s+(.+)$/i);
  if (!m) return null;
  const token = m[1].trim();
  if (!token) return null;

  const { data: session } = await supabaseAdmin
    .from('sg_sessions')
    .select('student_id, expires_at')
    .eq('token', token)
    .maybeSingle();

  if (!session) return null;
  if (new Date(session.expires_at).getTime() < Date.now()) return null;

  const { data: student } = await supabaseAdmin
    .from('sg_students')
    .select('*')
    .eq('id', session.student_id)
    .maybeSingle();

  return (student as Student) || null;
}

/**
 * نسخة صارمة: ترمي خطأ 401 إذا ما فيه طالب
 */
export async function requireStudent(req: Request): Promise<Student> {
  const student = await getStudentFromRequest(req);
  if (!student) {
    throw Object.assign(new Error('يجب تسجيل الدخول'), { status: 401 });
  }
  return student;
}

/** ملء الشاشة مع دعم webkit (Safari) وكشف الدعم. */
import { useCallback, useEffect, useState } from 'react';

type FsElement = HTMLElement & {
  webkitRequestFullscreen?: () => Promise<void> | void;
};

type FsDocument = Document & {
  webkitFullscreenElement?: Element | null;
  webkitExitFullscreen?: () => Promise<void> | void;
  webkitFullscreenEnabled?: boolean;
};

export function useFullscreen<T extends HTMLElement>() {
  const [el, setEl] = useState<T | null>(null);
  const [isFullscreen, setIsFullscreen] = useState(false);
  const [supported, setSupported] = useState(false);

  useEffect(() => {
    if (typeof document === 'undefined') return;
    const d = document as FsDocument;
    setSupported(Boolean(document.fullscreenEnabled || d.webkitFullscreenEnabled));
  }, []);

  useEffect(() => {
    if (typeof document === 'undefined') return;
    const d = document as FsDocument;
    const onChange = (): void => {
      const cur = document.fullscreenElement ?? d.webkitFullscreenElement ?? null;
      setIsFullscreen(cur !== null && cur === el);
    };
    document.addEventListener('fullscreenchange', onChange);
    document.addEventListener('webkitfullscreenchange', onChange);
    return () => {
      document.removeEventListener('fullscreenchange', onChange);
      document.removeEventListener('webkitfullscreenchange', onChange);
    };
  }, [el]);

  const enter = useCallback(async (): Promise<boolean> => {
    if (!el) return false;
    const target = el as FsElement;
    try {
      if (target.requestFullscreen) await target.requestFullscreen();
      else if (target.webkitRequestFullscreen) await target.webkitRequestFullscreen();
      else return false;
      return true;
    } catch {
      return false;
    }
  }, [el]);

  const exit = useCallback(async (): Promise<void> => {
    const d = document as FsDocument;
    try {
      if (document.exitFullscreen) await document.exitFullscreen();
      else if (d.webkitExitFullscreen) await d.webkitExitFullscreen();
    } catch {
      /* تجاهل */
    }
  }, []);

  const toggle = useCallback((): Promise<boolean | void> => {
    return isFullscreen ? exit() : enter();
  }, [isFullscreen, enter, exit]);

  return { setRef: setEl, isFullscreen, supported, enter, exit, toggle };
}

import { useEffect, useRef } from 'react';

export interface Shortcut {
  /** الحرف أو اسم المفتاح: 'k' أو '?' أو 'Escape' أو '1' */
  key: string;
  /** ⌘ أو Ctrl */
  meta?: boolean;
  alt?: boolean;
  /** إن لم يُحدَّد يُتجاهل Shift (مفيد لـ ؟ / ?) */
  shift?: boolean;
  /** يعمل أثناء الكتابة في حقل إدخال */
  allowInInput?: boolean;
  handler: () => void;
  description: string;
}

/** يطابق المفتاح الفعلي أو موضعه على الكيبورد، فيعمل مع التخطيط العربي (الحرف F = ب) */
function matchKey(e: KeyboardEvent, want: string): boolean {
  const w = want.toLowerCase();
  if (e.key.toLowerCase() === w) return true;
  if (/^[a-z]$/.test(w)) return e.code === `Key${w.toUpperCase()}`;
  if (/^[0-9]$/.test(w)) return e.code === `Digit${w}`;
  return false;
}

function match(e: KeyboardEvent, s: Shortcut): boolean {
  if (!matchKey(e, s.key)) return false;
  if (!!s.meta !== (e.metaKey || e.ctrlKey)) return false;
  if (!!s.alt !== e.altKey) return false;
  if (s.shift !== undefined && s.shift !== e.shiftKey) return false;
  return true;
}

function isEditable(el: EventTarget | null): boolean {
  if (!(el instanceof HTMLElement)) return false;
  const tag = el.tagName.toLowerCase();
  return tag === 'input' || tag === 'textarea' || tag === 'select' || el.isContentEditable;
}

export function useKeyboard(shortcuts: readonly Shortcut[], enabled = true): void {
  const ref = useRef(shortcuts);
  ref.current = shortcuts;

  useEffect(() => {
    if (!enabled) return;
    const onKey = (e: KeyboardEvent): void => {
      if (e.isComposing) return;
      for (const s of ref.current) {
        if (isEditable(e.target) && !s.allowInInput) continue;
        if (match(e, s)) {
          e.preventDefault();
          s.handler();
          return;
        }
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [enabled]);
}

import { useCallback, useEffect, useRef, useState } from 'react';
import { readRaw, subscribeRaw, writeRaw } from '@/lib/storage';

/**
 * حالة محفوظة في localStorage. القراءة بعد التركيب (آمنة مع SSR).
 * الكتابة تتم خارج دالة التحديث حتى لا تتكرر في StrictMode.
 */
export function useLocalState<T>(key: string, initial: T, version = 1): [T, (next: T | ((prev: T) => T)) => void] {
  const initRef = useRef(initial);
  const valueRef = useRef<T>(initial);
  const [value, setValue] = useState<T>(initial);

  useEffect(() => {
    const load = (): void => {
      const v = readRaw(key, initRef.current, version);
      valueRef.current = v;
      setValue(v);
    };
    load();
    return subscribeRaw(key, load);
  }, [key, version]);

  const set = useCallback(
    (next: T | ((prev: T) => T)) => {
      const resolved = typeof next === 'function' ? (next as (p: T) => T)(valueRef.current) : next;
      valueRef.current = resolved;
      setValue(resolved);
      writeRaw(key, resolved, version);
    },
    [key, version],
  );

  return [value, set];
}

// lib/storage.ts
/** تخزين محلي آمن: يتحمّل الوضع الخاص، امتلاء الحصة، وSSR. يدعم الاشتراك بين التبويبات. */

interface Entry<T> {
  version: number;
  data: T;
}

const PREFIX = 'lawazem:';
const EVENT = 'lawazem:storage';

function ok(): boolean {
  return typeof window !== 'undefined' && typeof localStorage !== 'undefined';
}

export function readRaw<T>(key: string, fallback: T, version = 1): T {
  if (!ok()) return fallback;
  try {
    const raw = localStorage.getItem(PREFIX + key);
    if (!raw) return fallback;
    const parsed = JSON.parse(raw) as Entry<T>;
    if (parsed && typeof parsed === 'object' && 'version' in parsed && 'data' in parsed) {
      return parsed.version === version ? parsed.data : fallback;
    }
    return fallback;
  } catch {
    return fallback;
  }
}

export function writeRaw<T>(key: string, value: T, version = 1): void {
  if (!ok()) return;
  try {
    const entry: Entry<T> = { version, data: value };
    localStorage.setItem(PREFIX + key, JSON.stringify(entry));
    window.dispatchEvent(new CustomEvent(EVENT, { detail: { key } }));
  } catch {
    /* الحصة ممتلئة أو التخزين معطّل */
  }
}

export function subscribeRaw(key: string, cb: () => void): () => void {
  if (!ok()) return () => {};
  const onStorage = (e: StorageEvent): void => {
    if (e.key === PREFIX + key) cb();
  };
  const onLocal = (e: Event): void => {
    if ((e as CustomEvent<{ key: string }>).detail?.key === key) cb();
  };
  window.addEventListener('storage', onStorage);
  window.addEventListener(EVENT, onLocal);
  return () => {
    window.removeEventListener('storage', onStorage);
    window.removeEventListener(EVENT, onLocal);
  };
}

// ==================== Approve ====================
export async function handleApprove(
  cb: {
    id: string;
    from: { id: number };
    message?: { chat: { id: number }; message_id: number };
  },
  id: string,
  admin: TelegramAdmin,
): Promise<void> {
  if (admin.role !== 'super_admin') {
    await answerCallbackQuery(cb.id, '❌ غير مصرح — تحتاج صلاحية مشرف أعلى');
    return;
  }

  const chatId = cb.message?.chat.id ?? 0;
  const messageId = cb.message?.message_id ?? 0;
  if (!chatId) return;

  const supabase = getSupabaseAdmin();

  const { data: existing } = await supabase
    .from('telegram_announcements')
    .select('status')
    .eq('id', id)
    .maybeSingle<{ status: string }>();

  if (!existing) {
    await answerCallbackQuery(cb.id, 'التبليغ غير موجود');
    return;
  }
  if (existing.status !== 'pending') {
    await answerCallbackQuery(cb.id, 'تمت معالجة هذا التبليغ مسبقاً');
    return;
  }

  await answerCallbackQuery(cb.id, '⏳ جاري الإرسال...');

  await supabase
    .from('telegram_announcements')
    .update({
      status: 'approved',
      reviewed_at: new Date().toISOString(),
      reviewed_by: 'super_admin',
    })
    .eq('id', id);

  let result: Awaited<ReturnType<typeof sendAnnouncement>>;
  try {
    result = await sendAnnouncement(id);
  } catch (err) {
    const msg = err instanceof Error ? err.message : 'خطأ غير معروف';
    await supabase
      .from('telegram_announcements')
      .update({ status: 'failed', sent_error: msg })
      .eq('id', id);
    await sendMessage(chatId, `❌ فشل الإرسال: ${escapeHtml(msg)}`);
    return;
  }

  if (result.error) {
    await supabase
      .from('telegram_announcements')
      .update({
        status: 'failed',
        sent_error: result.error,
        sent_count: result.sent,
      })
      .eq('id', id);
    await sendMessage(chatId, `❌ فشل الإرسال: ${escapeHtml(result.error)}`);
    return;
  }

  await supabase
    .from('telegram_announcements')
    .update({
      status: 'sent',
      sent_at: new Date().toISOString(),
      sent_count: result.sent,
      sent_error: result.failed > 0 ? `فشل ${result.failed}` : null,
    })
    .eq('id', id);

  if (messageId) {
    await editMessageText(chatId, messageId, '✅ تم الإرسال', {
      reply_markup: { inline_keyboard: [] },
    });
  }

  await sendMessage(
    chatId,
    `✅ <b>تم الإرسال</b>\n\n📤 نجح: ${result.sent}\n❌ فشل: ${result.failed}`,
  );
}

export async function handleRejectText(
  chatId: number,
  text: string,
  admin: TelegramAdmin,
): Promise<void> {
  if (admin.role !== 'super_admin') {
    await clearSession(chatId);
    return;
  }

  const session = await getSession(chatId);
  if (!session || session.flow_type !== 'reject') {
    await clearSession(chatId);
    return;
  }

  const id = session.flow_data.announcement_id as string | undefined;
  if (!id) {
    await clearSession(chatId);
    return;
  }

  const trimmed = text.trim();
  const note = trimmed === '-' || trimmed === '' ? null : trimmed.slice(0, 500);

  const { error } = await getSupabaseAdmin()
    .from('telegram_announcements')
    .update({
      status: 'rejected',
      reviewed_at: new Date().toISOString(),
      reviewed_by: 'super_admin',
      reviewed_note: note,
    })
    .eq('id', id)
    .eq('status', 'pending');

  await clearSession(chatId);

  if (error) {
    await sendMessage(chatId, '❌ فشل الرفض. حاول مرة ثانية.');
    return;
  }

  await sendMessage(chatId, '✅ تم رفض التبليغ.');
}

function buildPreview(draft: AnnouncementDraft): string {
  const lines: string[] = [];
  lines.push('📢 <b>معاينة التبليغ</b>');
  lines.push('');
  lines.push(
    `🎯 المرحلة: <b>${
      draft.stage ? escapeHtml(draft.stage) : 'كل المراحل'

    lines.push('<i>(لا يوجد عنوان)</i>');
  }
  lines.push('━━━━━━━━━━━━━━━');
  lines.push('');
  if (draft.body) {
    lines.push(escapeHtml(draft.body));
  } else {
    lines.push('<i>(لا يوجد نص)</i>');
  }
  if (draft.link_url) {
    lines.push('');
    lines.push(`🔗 ${escapeHtml(draft.link_label || draft.link_url)}`);

export async function handleAnnounceTextInput(
  chatId: number,
  text: string,
  step: string,
): Promise<void> {
  const session = await getSession(chatId);
  if (!session || session.flow_type !== 'announce') {
    await clearSession(chatId);
    return;
  }

  const draft = { ...(session.flow_data.draft as AnnouncementDraft) };
  const trimmed = text.trim();

  if (step === 'edit_title') {
    if (trimmed.length < 3 || trimmed.length > 200) {
      await sendMessage(chatId, '❌ العنوان يجب أن يكون بين 3 و 200 حرف.');
      return;
    }
    draft.title = trimmed;
  } else if (step === 'edit_body') {
    if (trimmed.length < 5 || trimmed.length > 2000) {
      await sendMessage(chatId, '❌ النص يجب أن يكون بين 5 و 2000 حرف.');
      return;
    }
    draft.body = trimmed;
  } else if (step === 'edit_link_url') {
    if (trimmed === '-') {
      draft.link_url = null;
      draft.link_label = null;
    } else {
      try {
        const u = new URL(trimmed);
        if (u.protocol !== 'http:' && u.protocol !== 'https:') throw new Error();
      } catch {
        await sendMessage(
          chatId,
          '❌ رابط غير صالح. يجب أن يبدأ بـ https://',
        );
        return;
      }
      draft.link_url = trimmed;
    }
  }

  await setSession(chatId, 'announce', 'draft', { draft });
  await sendMessage(chatId, buildPreview(draft), {
    reply_markup: draftPreviewKeyboard(),
  });
}

// ==================== Admin: Submit Reply ====================
export async function handleInquiryReplyText(
  chatId: number,
  text: string,
  adminChatId: number,
): Promise<void> {
  const session = await getSession(chatId);
  if (!session || session.flow_type !== 'inquiry') {
    await clearSession(chatId);
    return;
  }

  const inquiryId = session.flow_data.inquiry_id as string | undefined;
  if (!inquiryId) {
    await clearSession(chatId);
    await sendMessage(
      chatId,
      '❌ لم أجد الاستفسار، حاول مرة ثانية من القائمة.',
    );
    return;
  }

  const result = await answerInquiry({
    id: inquiryId,
    reply: text,
    adminChatId,
  });

  await clearSession(chatId);

  if (!result.ok || !result.inquiry) {
    await sendMessage(
      chatId,
      `❌ فشل الحفظ: ${result.error ?? 'غير معروف'}`,
    );
    return;
  }

  const inq = result.inquiry;

  // إرسال الرد للطالب
  const studentLines: string[] = [];
  studentLines.push('📬 <b>ردّ الإدارة على استفسارك</b>');
  studentLines.push('');
  studentLines.push('❓ <b>سؤالك:</b>');
  studentLines.push(escapeHtml(inq.question));
  studentLines.push('');
  studentLines.push('💬 <b>الرد:</b>');
  studentLines.push(escapeHtml(text));

  try {
    await sendMessage(inq.chat_id, studentLines.join('\n'), {
      reply_markup: STUDENT_MENU,
    });
  } catch {
    await sendMessage(
      chatId,
      '⚠️ تم الحفظ لكن تعذّر إرسال الرد للطالب.',
    );
    return;
  }

  await sendMessage(chatId, '✅ تم إرسال الرد للطالب.');
}

export async function handleInquiryClose(
  cb: {
    id: string;
    message?: { chat: { id: number }; message_id: number };
  },
  inquiryId: string,
): Promise<void> {
  const chatId = cb.message?.chat.id ?? 0;
  if (!chatId) return;

  const result = await closeInquiry(inquiryId);
  await answerCallbackQuery(
    cb.id,
    result.ok ? 'تم الإغلاق' : result.error ?? 'فشل',
  );

  if (result.ok && result.inquiry) {
    // إشعار الطالب بأن استفساره تم الاطلاع عليه
    try {
      await sendMessage(
        result.inquiry.chat_id,
        `📭 تم الاطلاع على استفسارك وإغلاقه. إن كان لديك سؤال آخر، أرسل استفساراً جديداً.`,
        { reply_markup: STUDENT_MENU },
      );
    } catch {
      /* ignore */
    }

    await sendMessage(chatId, '✅ تم إغلاق الاستفسار.');
  }
}

// ==================== /start ====================
export async function handleStart(
  msg: TelegramMessage,
  baseUrl: string,
): Promise<void> {
  if (!msg.from) return;
  const chatId = msg.chat.id;
  const firstName = msg.from.first_name ?? '';

  const existing = await getSubscriber(chatId);

  // ✅ مسجّل بالكامل — لا تطلب منه المرحلة من جديد
  if (isFullyOnboarded(existing) && existing) {
    const subjectCount = existing.subjects.length;
    const groupLine = `${groupEmoji(existing.group_name!)} <b>${groupLabel(
      existing.group_name!,
    )}</b>`;

    const lines: string[] = [];
    lines.push(
      `👋 <b>أهلاً من جديد${firstName ? `، ${escapeHtml(firstName)}` : ''}!</b>`,
    );
    lines.push('');
    lines.push(`📌 مرحلتك: <b>${escapeHtml(existing.stage!)}</b>`);
    lines.push(`👥 كروبك: ${groupLine}`);
    lines.push(
      `📚 موادك: <b>${subjectCount > 0 ? `${subjectCount} مادة` : 'الكل'}</b>`,
    );
    lines.push('');
    lines.push('👇 استخدم الأزرار بالأسفل للتنقل.');

    await sendMessage(chatId, lines.join('\n'), {
      reply_markup: STUDENT_MENU,
    });
    return;
  }

  // ===== مسجّل جزئياً أو جديد → ابدأ التسجيل =====
  await upsertSubscriber(msg.from, { onboarding_step: 'choosing_stage' });

  const greeting = firstName ? `، ${escapeHtml(firstName)}` : '';
  const text = existing
    ? `👋 أهلاً من جديد${greeting}!\n\nاختر مرحلتك الدراسية:`
    : `👋 أهلاً${greeting}!\n\nأنا <b>بوت لوازم</b> — سأخبرك بالجديد في مرحلتك مباشرة.\n\nاختر مرحلتك الدراسية:`;

  await sendMessage(chatId, text, { reply_markup: stageKeyboard() });
}
